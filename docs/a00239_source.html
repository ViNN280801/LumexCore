990  std::vector<T>
991  get_sequence (std::size_t count, T from_val = get_default_min (),
992  T to_val = get_default_max (),
993  DistributionType dist_type = DistributionType::UNIFORM) const
994  {
995  if (count == 0)
996  {
997  // Note: Returning empty sequence for zero count request
998  return {};
999  }
1000 
1001  try
1002  {
1003  std::vector<T> sequence;
1004  sequence.reserve (count); // Pre-allocate memory to avoid reallocations
1005 
1006  for (std::size_t i = 0; i < count; ++i)
1007  sequence.push_back (generate_number (from_val, to_val, dist_type));
1008 
1009  return sequence;
1010  }
1011  catch (std::exception const &e)
1012  {
1013  std::cerr << "Error generating sequence: " << e.what ()
1014  << ". Parameters: count=" << count << ", from=" << from_val
1015  << ", to=" << to_val << ". Returning empty sequence.\n";
1016  return {};
1017  }
1018  catch (...)
1019  {
1020  std::cerr << "Unknown exception during sequence generation. "
1021  << "Parameters: count=" << count << ", from=" << from_val
1022  << ", to=" << to_val << ". Returning empty sequence.\n";
1023  return {};
1024  }
1025  }
