134// Format an address as uppercase hexadecimal digits, no leading zeros/padding
135// (e.g. "1A2B3C"), matching the "[0x...]" style used throughout this file.
136// NOTE: fmt is NOT a declared dependency anywhere else in LumexLib (no
137// fmt::format usage/link in the rest of the tree), so this intentionally
138// avoids relying on it - std::ostringstream + std::hex gives the exact same
139// output without adding a new external dependency.
140static inline std::string
141formatHex (std::uintptr_t value)
142{
143 std::ostringstream oss;
144 oss << std::hex << std::uppercase << value;
145 return oss.str ();
146}

150// Resolve address to function name using addr2line (POSIX fallback)
151static inline std::string
152resolveAddressWithAddr2line (void *addr, char const *exe_path) LUMEX_NOEXCEPT
153{
154 try
155 {
156 // Create pipe for addr2line output
157 std::array<char, 512> buffer{};
158 std::string result;
159
160 // Build command: addr2line -C -f -e <executable> <address>
161 std::string cmd = lumex::core::string::utility::stringify (
162 "addr2line -C -f -e ", exe_path, " 0x",
163 formatHex (reinterpret_cast<uintptr_t> (addr)), " 2>/dev/null");
164
165 FILE *pipe = popen (cmd.c_str (),
166 "r"); // NOLINT(cppcoreguidelines-owning-memory)
167 if (!pipe)
168 return "";
169
170 // Read function name (first line)
171 if (fgets (buffer.data (), buffer.size (), pipe) != nullptr)
172 {
173 result = buffer.data ();
174 // Remove newline
175 if (!result.empty () && result.back () == '\n')
176 result.pop_back ();
177
178 // Read file:line (second line) and append
179 if (fgets (buffer.data (), buffer.size (), pipe) != nullptr)
180 {
181 std::string location = buffer.data ();
182 if (!location.empty () && location.back () == '\n')
183 location.pop_back ();
184
185 // Filter out "??:0" and "??:?"
186 if (location != "??:0" && location != "??:?")
187 result += " at " + location;
188 }
189 }
190
191 pclose (pipe); // NOLINT(cppcoreguidelines-owning-memory)
192
193 // Return empty if addr2line failed (contains "?" or "0x")
194 if (result.find ("??") != std::string::npos || result.empty ())
195 return "";
196
197 return result;
198 }
199 catch (...)
200 {
201 return "";
202 }
203}

239// Get executable directory for PDB search path.
240// NOTE: intentionally self-contained (does not use
241// lumex::core::filesystem::fs::lumex_filesystem::get_exe_path()):
242// lumex/core/filesystem publicly depends on lumex/core/utility (see its
243// CMakeLists.txt), so utility must not depend back on filesystem - that would
244// create a circular module dependency in the build graph.
245static inline std::string
246getExeDirectory () LUMEX_NOEXCEPT
247{
248 try
249 {
250 std::array<char, 4096> buf{};
251 DWORD const len = GetModuleFileNameA (nullptr, buf.data (),
252 static_cast<DWORD> (buf.size ()));
253 if (len == 0 || len >= buf.size ())
254 return {};
255 std::string path (buf.data (), len);
256 std::size_t const lastSlash = path.find_last_of ("\\/");
257 return (lastSlash != std::string::npos) ? path.substr (0, lastSlash + 1)
258 : std::string{};
259 }
260 catch (...)
261 {
262 return {};
263 }
264}

266// Lazy initialization for SymInitialize (amortize overhead)
267static inline bool
268ensureSymbolsInitialized (HANDLE process) LUMEX_NOEXCEPT
269{
270 static std::atomic_bool initialized{ false };
271 static std::once_flag init_flag;
272
273 if (initialized.load (std::memory_order_acquire))
274 return true;
275
276 std::call_once (init_flag, [process] () {
277 std::lock_guard<std::mutex> lock (getDbgHelpMutex ());
278 // SymSetOptions MUST be called BEFORE SymInitialize
279 // (MSDN)
280 SymSetOptions (SYMOPT_LOAD_LINES | SYMOPT_UNDNAME | SYMOPT_DEFERRED_LOADS);
281 std::string const exeDir = getExeDirectory ();
282 char const *searchPath = exeDir.empty () ? nullptr : exeDir.c_str ();
283 if (SymInitialize (process, searchPath, TRUE))
284 {
285 if (!exeDir.empty ())
286 SymSetSearchPath (process, exeDir.c_str ());
287 initialized.store (true, std::memory_order_release);
288 }
289 });
290
291 return initialized.load (std::memory_order_acquire);
292}

378static inline std::string
379captureStackTrace (int skip_frames = 1, int max_frames = 16) LUMEX_NOEXCEPT
380{
381 LUMEX_CONSTEXPR int kMaxStackFrames = 64;
382 std::string result;
383
384 try
385 {
386#if LUMEX_OS_WINDOWS
387 // Windows: DbgHelp API for symbol information
388 void *stack[kMaxStackFrames];
389
390 HANDLE process = GetCurrentProcess ();
391
392 // Thread-safe symbol initialization (lazy, once)
393 if (!Detail::ensureSymbolsInitialized (process))
394 {
395 // Fallback: only addresses without symbols
396 WORD frames = CaptureStackBackTrace (
397 static_cast<DWORD> (skip_frames),
398 static_cast<DWORD> (max_frames), stack, nullptr);
399
400 for (WORD i = 0; i < frames; ++i)
401 result += lumex::core::string::utility::stringify (
402 " #", i, ": [0x",
403 Detail::formatHex (reinterpret_cast<uintptr_t> (stack[i])),
404 "]\n");
405 return result.empty ()
406 ? " Stack trace unavailable (SymInitialize failed)\n"
407 : result;
408 }
409
410 // Capture stack with full symbol resolution
411 WORD frames = CaptureStackBackTrace (static_cast<DWORD> (skip_frames),
412 static_cast<DWORD> (max_frames),
413 stack, nullptr);
414
415 if (frames == 0)
416 return " Stack trace empty (no frames captured)\n";
417
418 // Allocate symbol buffer (variable size structure)
419 LUMEX_CONSTEXPR std::size_t kSymbolBufferSize
420 = sizeof (SYMBOL_INFO) + (MAX_SYM_NAME * sizeof (TCHAR));
421 auto *symbol_buffer
422 = static_cast<SYMBOL_INFO *> (malloc (kSymbolBufferSize));
423
424 if (symbol_buffer == nullptr)
425 return " Stack trace unavailable (memory allocation failed)\n";
426
427 symbol_buffer->MaxNameLen = MAX_SYM_NAME;
428 symbol_buffer->SizeOfStruct = sizeof (SYMBOL_INFO);
429
430 // Thread-safe symbol resolution
431 std::lock_guard<std::mutex> lock (Detail::getDbgHelpMutex ());
432
433 // Refresh module list to include DLLs loaded after SymInitialize (e.g.
434 // plugins loaded at runtime). Ensures SymFromAddr can resolve addresses
435 // in dynamically loaded modules.
436 SymRefreshModuleList (process);
437
438 for (WORD i = 0; i < frames; ++i)
439 {
440 DWORD64 address = reinterpret_cast<DWORD64> (stack[i]);
441
442 // Get function name
443 DWORD64 displacement = 0;
444 if (SymFromAddr (process, address, &displacement, symbol_buffer))
445 {
446 // Try to get line number and file name
447 IMAGEHLP_LINE64 line{};
448 line.SizeOfStruct = sizeof (IMAGEHLP_LINE64);
449 DWORD line_displacement = 0;
450
451 if (SymGetLineFromAddr64 (process, address, &line_displacement,
452 &line))
453 {
454 // Full info: function + file:line + address
455 result += lumex::core::string::utility::stringify (
456 " #", i, ": ", symbol_buffer->Name, " (", line.FileName,
457 ":", line.LineNumber, ") [0x",
458 Detail::formatHex (address), "]\n");
459 }
460 else
461 {
462 // Only function name + address (no line info)
463 result += lumex::core::string::utility::stringify (
464 " #", i, ": ", symbol_buffer->Name, " [0x",
465 Detail::formatHex (address), "]\n");
466 }
467 }
468 else
469 {
470 // Fallback: try SymGetModuleInfo64 for module name (e.g.
471 // ntdll.dll+0x1234)
472 IMAGEHLP_MODULE64 moduleInfo{};
473 moduleInfo.SizeOfStruct = sizeof (IMAGEHLP_MODULE64);
474 if (SymGetModuleInfo64 (process, address, &moduleInfo))
475 {
476 DWORD64 const offset = address - moduleInfo.BaseOfImage;
477 result += lumex::core::string::utility::stringify (
478 " #", i, ": ", moduleInfo.ModuleName, "+0x",
479 Detail::formatHex (offset), " [0x",
480 Detail::formatHex (address), "]\n");
481 }
482 else
483 {
484 result += lumex::core::string::utility::stringify (
485 " #", i, ": [0x", Detail::formatHex (address), "]\n");
486 }
487 }
488 }
489
490 free (
491 symbol_buffer); // NOLINT(cppcoreguidelines-owning-memory,cppcoreguidelines-no-malloc)
492
493#elif defined(__GNUC__) || defined(__clang__)
494 // Linux/POSIX: backtrace() + dladdr() for symbols
495 // NOTE: backtrace() - GNU extension, not in POSIX.1-2024, but widely
496 // available
497
498#if __has_include(<execinfo.h>)
499 void *addresses[kMaxStackFrames];
500
501 int frame_count = backtrace (addresses, kMaxStackFrames);
502
503 if (frame_count <= skip_frames)
504 return " Stack trace empty (insufficient frames)\n";
505
506 int actual_frames = std::min (frame_count - skip_frames, max_frames);
507
508 // Use backtrace_symbols for quick resolution (allocates memory)
509 char **symbols
510 = backtrace_symbols (addresses + skip_frames, actual_frames);
511
512 if (!symbols)
513 {
514 // Fallback: dladdr() for each address
515 for (int i = 0; i < actual_frames; ++i)
516 {
517 Dl_info info;
518 if (dladdr (addresses[i + skip_frames], &info))
519 {
520 std::string demangled
521 = Detail::demangleSymbol (info.dli_sname);
522 result += lumex::core::string::utility::stringify (
523 " #", i, ": ", demangled, " [0x",
524 Detail::formatHex (reinterpret_cast<uintptr_t> (
525 addresses[i + skip_frames])),
526 "]\n");
527 }
528 else
529 {
530 result += lumex::core::string::utility::stringify (
531 " #", i, ": [0x",
532 Detail::formatHex (reinterpret_cast<uintptr_t> (
533 addresses[i + skip_frames])),
534 "]\n");
535 }
536 }
537 return result;
538 }
539
540 // Get executable path for addr2line fallback
541 static std::string const exe_path = Detail::getExecutablePath ();
542 bool const use_addr2line_fallback = !exe_path.empty ();
543
544 // Process each frame with demangling
545 for (int i = 0; i < actual_frames; ++i)
546 {
547 Dl_info info;
548 bool has_symbol
549 = dladdr (addresses[i + skip_frames], &info) && info.dli_sname;
550 if (has_symbol)
551 {
552 // Demangle C++ symbol
553 std::string demangled = Detail::demangleSymbol (info.dli_sname);
554
555 // Calculate offset within function
556 ptrdiff_t offset
557 = reinterpret_cast<char *> (addresses[i + skip_frames])
558 - reinterpret_cast<char *> (info.dli_saddr);
559
560 result += lumex::core::string::utility::stringify (
561 " #", i, ": ", demangled, " +", offset, " [0x",
562 Detail::formatHex (reinterpret_cast<uintptr_t> (
563 addresses[i + skip_frames])),
564 "]\n");
565 }
566 else
567 {
568 // Try addr2line fallback if no symbol info available
569 std::string addr2line_result;
570 if (use_addr2line_fallback)
571 addr2line_result = Detail::resolveAddressWithAddr2line (
572 addresses[i + skip_frames], exe_path.c_str ());
573
574 if (!addr2line_result.empty ())
575 {
576 // addr2line succeeded - use its output
577 result += lumex::core::string::utility::stringify (
578 " #", i, ": ", addr2line_result, " [0x",
579 Detail::formatHex (reinterpret_cast<uintptr_t> (
580 addresses[i + skip_frames])),
581 "]\n");
582 }
583 else
584 {
585 // Complete fallback: raw backtrace_symbols output
586 result += lumex::core::string::utility::stringify (
587 " #", i, ": ", symbols[i], " [0x",
588 Detail::formatHex (reinterpret_cast<uintptr_t> (
589 addresses[i + skip_frames])),
590 "]\n");
591 }
592 }
593 }
594
595 free (
596 symbols); // NOLINT(cppcoreguidelines-owning-memory,cppcoreguidelines-no-malloc)
597
598#else
599 // No backtrace support - return basic info
600 result = " Stack trace unavailable (<execinfo.h> not available)\n";
601 result += lumex::core::string::utility::stringify (
602 " Current function: ", LUMEX_FUNCTION_NAME, "\n");
603#endif
604
605#else
606 // Unknown platform
607 result = " Stack trace not supported on this platform\n";
608 result += lumex::core::string::utility::stringify (
609 " Current function: ", LUMEX_FUNCTION_NAME, "\n");
610#endif
611 }
612 catch (std::exception const &exc)
613 {
614 return lumex::core::string::utility::stringify (
615 " Stack trace unavailable (exception: ", exc.what (), ")\n");
616 }
617 catch (...)
618 {
619 return " Stack trace unavailable (unknown exception)\n";
620 }
621
622 return result.empty () ? " Stack trace empty\n" : result;
623}
