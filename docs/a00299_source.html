186measure_execution_time (Callable &&callable, Args &&...args)

188 LUMEX_STATIC_ASSERT_MSG (

215template <typename Callable>
216bool
217measure_time (Callable &&callable, std::string const &message,
218 std::ostream &out = std::clog, bool need_to_gate_via_env = true,
219 LUMEX_ATTRIBUTE_MAYBE_UNUSED char const *env_name
220 = default_measure_time_env_name ())
221{
222 LUMEX_STATIC_ASSERT_MSG (
223 lumex::core::utility::traits::invoke::is_callable_v<Callable>,
224 "Callable must be nullary-invocable");
225
226 if (need_to_gate_via_env)
227 {
228 if (env_name == nullptr
229 || !lumex::core::environment::env::is_env_set (env_name))
230 {
231 std::forward<Callable> (callable) ();
232 return false;
233 }
234 }
235 else
236 {
237 LUMEX_ATTRIBUTE_MAYBE_UNUSED_VAR (env_name);
238 }
239
240 long long const elapsed_ms
241 = measure_execution_time (std::forward<Callable> (callable));
242 write_measure_time_report (out, message, elapsed_ms);
243 return true;
244}
