131 T
132 generate_number(T from_val, T to_val, DistributionType dist_type) const
133 {
134 switch(dist_type)
135 {
136 case DistributionType::UNIFORM: return generate_uniform(from_val, to_val);
137
138 case DistributionType::NORMAL: return generate_normal_safe(from_val, to_val);
139
140 case DistributionType::EXPONENTIAL: return generate_exponential_safe(from_val);
141
142 case DistributionType::GAMMA: return generate_gamma_safe(from_val, to_val);
143
144 case DistributionType::BERNOULLI: return generate_bernoulli_safe(static_cast<double>(from_val));
145
146 case DistributionType::BINOMIAL:
147 return generate_binomial_safe(static_cast<int>(to_val), static_cast<double>(from_val));
148
149 case DistributionType::GEOMETRIC: return generate_geometric_safe(static_cast<double>(from_val));
150
151 case DistributionType::POISSON: return generate_poisson_safe(static_cast<double>(from_val));
152
153 default: return generate_uniform(from_val, to_val);
154 }
155 }

732 std::vector<T>
733 get_sequence(std::size_t count, T from_val = get_default_min(), T to_val = get_default_max(),
734 DistributionType dist_type = DistributionType::UNIFORM) const
735 {
736 if(count == 0)
737 {
738 // Note: Returning empty sequence for zero count request
739 return {};
740 }
741
742 try
743 {
744 std::vector<T> sequence;
745 sequence.reserve(count); // Pre-allocate memory to avoid reallocations
746
747 for(std::size_t i = 0; i < count; ++i) sequence.push_back(generate_number(from_val, to_val, dist_type));
748
749 return sequence;
750 }
751 catch(std::exception const &e)
752 {
753 std::cerr << "Error generating sequence: " << e.what() << ". Parameters: count=" << count << ", from=" << from_val
754 << ", to=" << to_val << ". Returning empty sequence.\n";
755 return {};
756 }
757 catch(...)
758 {
759 std::cerr << "Unknown exception during sequence generation. "
760 << "Parameters: count=" << count << ", from=" << from_val << ", to=" << to_val
761 << ". Returning empty sequence.\n";
762 return {};
763 }
764 }
