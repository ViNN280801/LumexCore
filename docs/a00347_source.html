880 {
881 // Check ranges to prevent overflow
882 // Signedness is taken into account so the comparison is correct
883 if ((std::is_signed<clean_T>::value && std::is_unsigned<clean_U>::value)
884 ? true
885 : false)
886 {
887 // T signed, U unsigned: if current < 0 then current >= other is never
888 // false
889 if (current < 0)
890 return false;
891 // other is always >= 0; only the upper bound is checked
892#ifdef LUMEXLUMEX_SFNC_ARCH_X86
893 using UnsignedT = typename std::make_unsigned<clean_T>::type;
894 if (static_cast<UnsignedT> (other)
895 > static_cast<UnsignedT> (std::numeric_limits<clean_T>::max ()))
896 return false;
897#else
898 if (other > static_cast<typename std::make_unsigned<clean_T>::type> (
899 std::numeric_limits<clean_T>::max ()))
900 return false;
901#endif
902 }

904 && std::is_signed<clean_U>::value)
905 ? true
906 : false)

908 // T unsigned, U signed: if other < 0 then current >= other is always
909 // true true
910 if (other < 0)
911 return true;

913#ifdef LUMEXLUMEX_SFNC_ARCH_X86
914 using UnsignedU = typename std::make_unsigned<clean_U>::type;
