413 template <class Comp>
414 static bool
415 compare_eq (
416 XPathAstNode *lhs, // NOLINT(readability-function-cognitive-complexity)
417 XPathAstNode *rhs, XPathContext const &ctx, XPathStack const &stack,
418 Comp const &comp)
419 {
420 xpath_value_type lt_ = lhs->rettype ();
421 xpath_value_type rt_ = rhs->rettype ();
422
423 if (lt_ != xpath_type_node_set && rt_ != xpath_type_node_set)
424 {
425 if (lt_ == xpath_type_boolean || rt_ == xpath_type_boolean)
426 return comp (lhs->eval_boolean (ctx, stack),
427 rhs->eval_boolean (ctx, stack));
428 if (lt_ == xpath_type_number || rt_ == xpath_type_number)
429 return comp (lhs->eval_number (ctx, stack),
430 rhs->eval_number (ctx, stack));
431 if (lt_ == xpath_type_string || rt_ == xpath_type_string)
432 {
433 XPathAllocatorCapture capture (stack.result);
434
435 XPathString ls_ = lhs->eval_string (ctx, stack);
436 XPathString rs_ = rhs->eval_string (ctx, stack);
437
438 return comp (ls_, rs_);
439 }
440 }
441 else if (lt_ == xpath_type_node_set && rt_ == xpath_type_node_set)
442 {
443 XPathAllocatorCapture capture (stack.result);
444
445 XPathNodeSetRaw ls_
446 = lhs->eval_node_set (ctx, stack, nodeset_eval_all);
447 XPathNodeSetRaw rs_
448 = rhs->eval_node_set (ctx, stack, nodeset_eval_all);
449
450 for (XPathNode const *li = ls_.begin (); li != ls_.end (); ++li)
451 for (XPathNode const *ri = rs_.begin (); ri != rs_.end (); ++ri)
452 {
453 XPathAllocatorCapture cri (stack.result);
454
455 if (comp (string_value (*li, stack.result),
456 string_value (*ri, stack.result)))
457 return true;
458 }
459
460 return false;
461 }
462 else
463 {
464 if (lt_ == xpath_type_node_set)
465 {
466 std::swap (lhs, rhs);
467 std::swap (lt_, rt_);
468 }
469
470 if (lt_ == xpath_type_boolean)
471 return comp (lhs->eval_boolean (ctx, stack),
472 rhs->eval_boolean (ctx, stack));
473 if (lt_ == xpath_type_number)
474 {
475 XPathAllocatorCapture capture (stack.result);
476
477 double tmp = lhs->eval_number (ctx, stack);
478 XPathNodeSetRaw rs_
479 = rhs->eval_node_set (ctx, stack, nodeset_eval_all);
480
481 for (XPathNode const *ri = rs_.begin (); ri != rs_.end (); ++ri)
482 {
483 XPathAllocatorCapture cri (stack.result);
484
485 if (comp (tmp, convert_string_to_number (
486 string_value (*ri, stack.result).c_str ())))
487 return true;
488 }
489
490 return false;
491 }
492 if (lt_ == xpath_type_string)
493 {
494 XPathAllocatorCapture capture (stack.result);
495
496 XPathString tmp = lhs->eval_string (ctx, stack);
497 XPathNodeSetRaw rs_

499
500 for (XPathNode const *ri = rs_.begin (); ri != rs_.end (); ++ri)

507 return false;
508 }

511 LUMEX_ASSERT (false && "Wrong types"); // unreachable
512 return false;

786 template <class T>
787 void
788 step_fill (XPathNodeSetRaw
789 &nsr, // NOLINT(readability-function-cognitive-complexity)
790 XmlNodeBase *node, XPathAllocator *alloc, bool once, T /*val*/)
791 {
792 axis_t const axis = T::axis;
793
794 switch (axis)
795 {
796 case axis_attribute:
797 {
798 for (XmlAttributeBase *attr = node->first_attribute; attr;
799 attr = attr->next_attribute)
800 if (step_push (nsr, attr, node, alloc) & once)
801 return;
802
803 break;
804 }
805
806 case axis_child:
807 {
808 for (XmlNodeBase *ctx = node->first_child; ctx;
809 ctx = ctx->next_sibling)
810 if (step_push (nsr, ctx, alloc) & once)
811 return;
812
813 break;
814 }
815
816 case axis_descendant:
817 case axis_descendant_or_self:
818 {
819 if (axis == axis_descendant_or_self)
820 if (step_push (nsr, node, alloc) & once)
821 return;
822
823 XmlNodeBase *cur = node->first_child;
824
825 while (cur)
826 {
827 if (step_push (nsr, cur, alloc) & once)
828 return;
829
830 if (cur->first_child)
831 cur = cur->first_child;
832 else
833 {
834 while (!cur->next_sibling)
835 {
836 cur = cur->parent;
837
838 if (cur == node)
839 return;
840 }
841
842 cur = cur->next_sibling;
843 }
844 }
845
846 break;
847 }
848
849 case axis_following_sibling:
850 {
851 for (XmlNodeBase *ctx = node->next_sibling; ctx;
852 ctx = ctx->next_sibling)
853 if (step_push (nsr, ctx, alloc) & once)
854 return;
855
856 break;
857 }
858
859 case axis_preceding_sibling:
860 {
861 for (XmlNodeBase *ctx = node->prev_sibling_c; ctx->next_sibling;
862 ctx = ctx->prev_sibling_c)
863 if (step_push (nsr, ctx, alloc) & once)
864 return;
865
866 break;
867 }
868
869 case axis_following:
870 {
871 XmlNodeBase *cur = node;
872
873 // exit from this node so that we don't include descendants
874 while (!cur->next_sibling)
875 {
876 cur = cur->parent;
877
878 if (!cur)
879 return;
880 }
881
882 cur = cur->next_sibling;
883
884 while (cur)
885 {
886 if (step_push (nsr, cur, alloc) & once)
887 return;
888
889 if (cur->first_child)
890 cur = cur->first_child;
891 else
892 {
893 while (!cur->next_sibling)
894 {
895 cur = cur->parent;
896
897 if (!cur)
898 return;
899 }
900
901 cur = cur->next_sibling;
902 }
903 }
904
905 break;
906 }
907
908 case axis_preceding:
909 {
910 XmlNodeBase *cur = node;
911
912 // exit from this node so that we don't include descendants
913 while (!cur->prev_sibling_c->next_sibling)
914 {
915 cur = cur->parent;
916
917 if (!cur)
918 return;
919 }
920
921 cur = cur->prev_sibling_c;
922
923 while (cur)
924 {
925 if (cur->first_child)
926 cur = cur->first_child->prev_sibling_c;
927 else
928 {
929 // leaf node, can't be ancestor
930 if (step_push (nsr, cur, alloc) & once)
931 return;
932
933 while (!cur->prev_sibling_c->next_sibling)
934 {
935 cur = cur->parent;
936
937 if (!cur)
938 return;
939
940 if (!node_is_ancestor (cur, node))
941 if (step_push (nsr, cur, alloc) & once)
942 return;
943 }
944
945 cur = cur->prev_sibling_c;
946 }
947 }
948
949 break;
950 }
951
952 case axis_ancestor:
953 case axis_ancestor_or_self:
954 {
955 if (axis == axis_ancestor_or_self)
956 if (step_push (nsr, node, alloc) & once)
957 return;
958
959 XmlNodeBase *cur = node->parent;
960
961 while (cur)
962 {
963 if (step_push (nsr, cur, alloc) & once)
964 return;
965
966 cur = cur->parent;
967 }
968
969 break;
970 }
971
972 case axis_self:
973 {
974 step_push (nsr, node, alloc);
975
976 break;
977 }
978
979 case axis_parent:
980 {
981 if (node->parent)
982 step_push (nsr, node->parent, alloc);
983
984 break;
985 }
986
987 default:
988 LUMEX_ASSERT (false && "Unimplemented axis"); // unreachable
989 }
990 }

1069

1078 return;

1081 break;

1088 break;

1158 XPathNodeSetRaw
1159 step_do (XPathContext const &ctx, XPathStack const &stack,
1160 Types::nodeset_eval_t eval, T val)
1161 {
1162 axis_t const axis = T::axis;
1163 bool const axis_reverse
1164 = (axis == axis_ancestor || axis == axis_ancestor_or_self
1165 || axis == axis_preceding || axis == axis_preceding_sibling);
1166 XPathNodeSet::type_t const axis_type
1167 = axis_reverse ? XPathNodeSet::type_sorted_reverse
1168 : XPathNodeSet::type_sorted;
1169
1170 bool once = (axis == axis_attribute && m_test == nodetest_name)
1171 || (!m_right && eval_once (axis_type, eval)) ||
1172 // coverity[mixed_enums]
1173 (m_right && !m_right->m_next
1174 && m_right->m_test == predicate_constant_one);
1175
1176 XPathNodeSetRaw nsr;
1177 nsr.set_type (axis_type);
1178
1179 if (m_left)
1180 {
1181 XPathNodeSetRaw tmp
1182 = m_left->eval_node_set (ctx, stack, nodeset_eval_all);
1183
1184 // self axis preserves the original order
1185 if (axis == axis_self)
1186 nsr.set_type (tmp.type ());
1187
1188 for (XPathNode const *it = tmp.begin (); it != tmp.end (); ++it)
1189 {
1190 std::size_t size = nsr.size ();
1191
1192 // in general, all axes generate elements in a particular order,
1193 // but there is no order guarantee if axis is applied to two nodes
1194 if (axis != axis_self && size != 0)
1195 nsr.set_type (XPathNodeSet::type_unsorted);
1196
1197 step_fill (nsr, *it, stack.result, once, val);
1198 if (m_right)
1199 apply_predicates (nsr, size, stack, eval);
1200 }
1201 }
1202 else
1203 {
1204 step_fill (nsr, ctx.node, stack.result, once, val);
1205 if (m_right)
1206 apply_predicates (nsr, 0, stack, eval);
1207 }
1208
1209 // child, attribute and self axes always generate unique set of nodes
1210 // for other axis, if the set stayed sorted, it stayed unique because the
1211 // traversal algorithms do not visit the same node twice
1212 if (axis != axis_child && axis != axis_attribute && axis != axis_self
1213 && nsr.type () == XPathNodeSet::type_unsorted)
1214 nsr.remove_duplicates (stack.temp);
1215
1216 return nsr;
1217 }
