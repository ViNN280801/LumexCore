112 template <typename SuccessType, typename ErrorType> class Expected

114 public:
115  // ====================== Static assertions ====================== //
116  // References are forbidden because:
117  // 1. This class (and its specializations) is meant to
118  // own the stored success or error value. If SuccessType or ErrorType were
119  // references, for example `int&`, Expected would not own the object, only
120  // refer to it. That can dangle if the original object referred to by
121  // SuccessType (or ErrorType) is destroyed before Expected.
122  // 2. Union restriction: the C++ standard forbids reference members in a
123  // `union`,
124  // because references are neither movable nor copyable.
125  // 3. C++23 compatibility: std::expected also forbids reference types for
126  // the error parameter for the same reasons. Behavior stays consistent and
127  // predictable.
128  LUMEX_STATIC_ASSERT_MSG (!std::is_reference<SuccessType>::value,
129  "Expected<T,E>: T must not be a reference");
130  LUMEX_STATIC_ASSERT_MSG (!std::is_function<SuccessType>::value,
131  "Expected<T,E>: T must not be a function type");
132  LUMEX_STATIC_ASSERT_MSG (
133  !std::is_same<typename std::decay<SuccessType>::type,
134  in_place_tag>::value,
135  "Expected<T,E>: T must not be in_place_tag");
136  LUMEX_STATIC_ASSERT_MSG (
137  !std::is_same<typename std::decay<SuccessType>::type, unexpect_t>::value,

539  LUMEX_ATTRIBUTE_NODISCARD (
540  "Return value is the contained error; should always be used.")
541  LUMEX_CONSTEXPR_FUNCTION ErrorType &
542  error () &
543  {
544  // Precondition: !m_has_value.
545  // In std::expected, error() is an unchecked observer: calling it without
546  // an error is undefined behavior, and checked access is value(), which
547  // throws bad_expected_access<E>. Here the storage is a union, so reading
548  // m_storage.m_error while m_has_value is true would read an inactive
549  // member, which is undefined behavior in C++.
550  //
551  // This method checks the precondition with LUMEX_ASSERT instead of
552  // throwing. LUMEX_ASSERT is active in every build, NDEBUG included: a
553  // violation prints the condition, the file and the line, and aborts the
554  // program before the inactive member is read. The check costs one branch
555  // in every build. Because error() never throws, it stays usable where an
556  // exception must not escape (destructors, swap, emergency paths);
557  // checked, catchable access is value() or a prior has_value() check.
558  LUMEX_ASSERT (!m_has_value
559  && "error() called on an Expected that holds a value");
560  return m_storage.m_error;
561  }
