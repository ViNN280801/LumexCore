278inline void
279copy_parsed_name (char *dest, std::size_t dest_size,
280 char const *pretty) LUMEX_NOEXCEPT
281{
282 std::size_t len = 0;
283 while (pretty[len] != '\0')
284 ++len;
285
286 // GCC/Clang pretty strings use `.member`; MSVC __FUNCSIG__ uses
287 // `->member` (`...storage<T>->value->id`). Prefer the last arrow, then
288 // the last dot. Falling back to `:` alone picks the aggregate type name
289 // on MSVC (`...::Plain,&...->value->id`) and is wrong.
290 std::size_t arrow = static_cast<std::size_t> (-1);
291 std::size_t dot = static_cast<std::size_t> (-1);
292 std::size_t colon = static_cast<std::size_t> (-1);
293 for (std::size_t i = 0; i < len; ++i)
294 {
295 if (pretty[i] == '.')
296 dot = i;
297 if (pretty[i] == ':')
298 colon = i;
299 if (i + 1 < len && pretty[i] == '-' && pretty[i + 1] == '>')
300 arrow = i;
301 }
302
303 std::size_t start = 0;
304 if (arrow != static_cast<std::size_t> (-1))
305 start = arrow + 2;
306 else if (dot != static_cast<std::size_t> (-1))
307 start = dot + 1;
308 else if (colon != static_cast<std::size_t> (-1))
309 start = colon + 1;
310
311 std::size_t out = 0;
312 while (start < len && is_ident_char (pretty[start]) && out + 1 < dest_size)
313 dest[out++] = pretty[start++];
314 dest[out] = '\0';
315}
