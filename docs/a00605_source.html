327 inline void
328 copy_parsed_name (char *dest, std::size_t dest_size,
329  char const *pretty) LUMEX_NOEXCEPT
330 {
331  std::size_t len = 0;
332  while (pretty[len] != '\0')
333  ++len;
334 
335  // GCC/Clang pretty strings use `.member`; MSVC __FUNCSIG__ uses
336  // `->member` (`...storage<T>->value->id`). Prefer the last arrow, then
337  // the last dot. Falling back to `:` alone picks the aggregate type name
338  // on MSVC (`...::Plain,&...->value->id`) and is wrong.
339  std::size_t arrow = static_cast<std::size_t> (-1);
340  std::size_t dot = static_cast<std::size_t> (-1);
341  std::size_t colon = static_cast<std::size_t> (-1);
342  for (std::size_t i = 0; i < len; ++i)
343  {
344  if (pretty[i] == '.')
345  dot = i;
346  if (pretty[i] == ':')
347  colon = i;
348  if (i + 1 < len && pretty[i] == '-' && pretty[i + 1] == '>')
349  arrow = i;
350  }
351 
352  std::size_t start = 0;
353  if (arrow != static_cast<std::size_t> (-1))
354  start = arrow + 2;
355  else if (dot != static_cast<std::size_t> (-1))
356  start = dot + 1;
357  else if (colon != static_cast<std::size_t> (-1))
358  start = colon + 1;
359 
360  std::size_t out = 0;
361  while (start < len && is_ident_char (pretty[start]) && out + 1 < dest_size)
362  dest[out++] = pretty[start++];
363  dest[out] = '\0';
364 }
