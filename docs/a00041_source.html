141 static bool
142 _type_list_contains (nlohmann::json const &type_field,
143 char const *name) LUMEX_NOEXCEPT
144 {
145 if (type_field.is_string ())
146 return type_field.get_ref<std::string const &> () == name;
147 if (type_field.is_array ())
148 {
149 for (nlohmann::json::const_iterator it = type_field.cbegin ();
150 it != type_field.cend (); ++it)
151 {
152 if (it->is_string ()
153 && it->get_ref<std::string const &> () == name)
154 return true;
155 }
156 }
157 return false;
158 }

167 static bool
168 _matches_leaf_type (nlohmann::json const &type_field,
169 nlohmann::json const &value)
170 {
171 if (_type_list_contains (type_field, "string") && value.is_string ())
172 return true;
173 if (_type_list_contains (type_field, "boolean") && value.is_boolean ())
174 return true;
175 if (_type_list_contains (type_field, "number") && value.is_number ())
176 return true;
177 if (_type_list_contains (type_field, "null") && value.is_null ())
178 return true;
179 if (_type_list_contains (type_field, "integer"))
180 {
181 if (value.is_number_integer ())
182 return true;
183 // Producers that compute a whole number in floating point still
184 // satisfy "integer" when the fractional part is zero.
185 if (value.is_number_float ())
186 {
187 double const number = value.get<double> ();
188 if (std::floor (number) == number)
189 return true;
190 }
191 }
192 return false;
193 }

216 static nlohmann::json
217 _traverse_leaf (nlohmann::json const &schema, nlohmann::json const &input,
218 std::string const &path, LumexJsonSchemaCheckMode mode)
219 {
220 if (input.is_null ())
221 return nullptr;
222
223 nlohmann::json const &type_field = schema.at ("type");
224 if (!_matches_leaf_type (type_field, input))
225 {
226 throw LumexJsonSchemaException (
227 LumexJsonSchemaFailure::type_mismatch, path,
228 "expected type " + type_field.dump () + ", got "
229 + input.type_name () + " (" + input.dump () + ")");
230 }
231
232 nlohmann::json::const_iterator const allowed = schema.find ("enum");
233 if (allowed != schema.cend () && !_contains_value (*allowed, input))
234 {
235 throw LumexJsonSchemaException (
236 LumexJsonSchemaFailure::enum_mismatch, path,
237 "expected one of " + allowed->dump () + ", got " + input.dump ());
238 }
239 return mode == LumexJsonSchemaCheckMode::validate_and_normalize
240 ? input
241 : nlohmann::json (nullptr);
242 }

244 static nlohmann::json
245 _traverse_object (nlohmann::json const &schema, nlohmann::json const &input,
246 std::string const &path, LumexJsonSchemaCheckMode mode)
247 {
248 if (!input.is_object ())
249 {
250 throw LumexJsonSchemaException (
251 LumexJsonSchemaFailure::missing_required, path,
252 std::string ("expected object, got ") + input.type_name ());
253 }
254
255 bool const build
256 = mode == LumexJsonSchemaCheckMode::validate_and_normalize;
257 nlohmann::json::const_iterator const properties
258 = schema.find ("properties");
259 if (properties == schema.cend ())
260 return build ? nlohmann::json::object () : nlohmann::json (nullptr);
261
262 std::vector<std::string> required;
263 nlohmann::json::const_iterator const required_field
264 = schema.find ("required");
265 if (required_field != schema.cend ())
266 {
267 for (nlohmann::json::const_iterator it = required_field->cbegin ();
268 it != required_field->cend (); ++it)
269 required.push_back (it->get<std::string> ());
270 }
271
272 nlohmann::json result
273 = build ? nlohmann::json::object () : nlohmann::json (nullptr);
274 for (nlohmann::json::const_iterator it = properties->cbegin ();
275 it != properties->cend (); ++it)
276 {
277 std::string const &name = it.key ();
278 nlohmann::json const &property_schema = it.value ();
279 std::string const child_path = path + "." + name;
280
281 nlohmann::json::const_iterator const child = input.find (name);
282 if (child != input.cend () && !child->is_null ())
283 {
284 nlohmann::json value
285 = _traverse_node (property_schema, *child, child_path, mode);
286 if (build)
287 result[name] = std::move (value);
288 continue;
289 }
290
291 if (!_contains_name (required, name))
292 continue; // optional and missing: omitted from the result
293
294 nlohmann::json::const_iterator const property_type
295 = property_schema.find ("type");
296 bool const composite = property_type != property_schema.cend ()
297 && _is_composite_type (*property_type);
298 if (composite)
299 {
300 // A required object/array whose own type also allows "null" may
301 // be absent or null: both normalize to null. Otherwise it is an
302 // error.
303 if (_type_list_contains (*property_type, "null"))
304 {
305 if (build)
306 result[name] = nullptr;
307 continue;
308 }
309 throw LumexJsonSchemaException (
310 LumexJsonSchemaFailure::missing_required, child_path,
311 "required object/array field is missing from input");
312 }
313
314 if (build)
315 result[name] = nullptr; // required leaf, missing: null
316 }
317 return result;
318 }

320 static nlohmann::json
321 _traverse_array (nlohmann::json const &schema, nlohmann::json const &input,
322 std::string const &path, LumexJsonSchemaCheckMode mode)
323 {
324 if (!input.is_array ())
325 {
326 throw LumexJsonSchemaException (
327 LumexJsonSchemaFailure::missing_required, path,
328 std::string ("expected array, got ") + input.type_name ());
329 }
330
331 bool const build
332 = mode == LumexJsonSchemaCheckMode::validate_and_normalize;
333 nlohmann::json::const_iterator const items = schema.find ("items");
334 if (items == schema.cend ())
335 return build ? input : nlohmann::json (nullptr); // untyped array
336
337 nlohmann::json result
338 = build ? nlohmann::json::array () : nlohmann::json (nullptr);
339 for (std::size_t index = 0; index < input.size (); ++index)
340 {
341 nlohmann::json value
342 = _traverse_node (*items, input[index],
343 path + "[" + std::to_string (index) + "]", mode);
344 if (build)
345 result.push_back (std::move (value));
346 }
347 return result;
348 }

350 static nlohmann::json
351 _traverse_node (nlohmann::json const &schema, nlohmann::json const &input,
352 std::string const &path, LumexJsonSchemaCheckMode mode)
353 {
354 nlohmann::json::const_iterator const constant = schema.find ("const");
355 if (constant != schema.cend ())
356 {
357 if (input.is_null ())
358 return nullptr;
359 if (input != *constant)
360 {
361 throw LumexJsonSchemaException (
362 LumexJsonSchemaFailure::const_mismatch, path,
363 "expected const " + constant->dump () + ", got "
364 + input.dump ());
365 }
366 return mode == LumexJsonSchemaCheckMode::validate_and_normalize
367 ? input
368 : nlohmann::json (nullptr);
369 }
370
371 nlohmann::json const &type_field = schema.at ("type");
372 if (_type_list_contains (type_field, "object"))
373 return _traverse_object (schema, input, path, mode);
374 if (_type_list_contains (type_field, "array"))
375 return _traverse_array (schema, input, path, mode);
376 return _traverse_leaf (schema, input, path, mode);
377 }
