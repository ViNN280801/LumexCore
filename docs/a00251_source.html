323 inline void
324 copy_parsed_name (char *dest, std::size_t dest_size,
325  char const *pretty) LUMEX_NOEXCEPT
326 {
327  std::size_t len = 0;
328  while (pretty[len] != '\0')
329  ++len;
330 
331  // GCC/Clang pretty strings use `.member`; MSVC __FUNCSIG__ uses
332  // `->member` (`...storage<T>->value->id`). Prefer the last arrow, then
333  // the last dot. Falling back to `:` alone picks the aggregate type name
334  // on MSVC (`...::Plain,&...->value->id`) and is wrong.
335  std::size_t arrow = static_cast<std::size_t> (-1);
336  std::size_t dot = static_cast<std::size_t> (-1);
337  std::size_t colon = static_cast<std::size_t> (-1);
338  for (std::size_t i = 0; i < len; ++i)
339  {
340  if (pretty[i] == '.')
341  dot = i;
342  if (pretty[i] == ':')
343  colon = i;
344  if (i + 1 < len && pretty[i] == '-' && pretty[i + 1] == '>')
345  arrow = i;
346  }
347 
348  std::size_t start = 0;
349  if (arrow != static_cast<std::size_t> (-1))
350  start = arrow + 2;
351  else if (dot != static_cast<std::size_t> (-1))
352  start = dot + 1;
353  else if (colon != static_cast<std::size_t> (-1))
354  start = colon + 1;
355 
356  std::size_t out = 0;
357  while (start < len && is_ident_char (pretty[start]) && out + 1 < dest_size)
358  dest[out++] = pretty[start++];
359  dest[out] = '\0';
360 }
