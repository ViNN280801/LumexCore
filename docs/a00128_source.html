152inline bool
153_is_base64_char_impl (Types::byte_type chr) LUMEX_NOEXCEPT
154{
155 // A character is a valid Base64 alphabet character if its decoded value is
156 // not kBase64DecodeInvalidChar. This function specifically excludes '=' as a
157 // valid alphabet character, as '=' is padding.
158 return _decode_table.at (chr) != Constants::kBase64DecodeInvalidChar;
159}

169template <typename T>
170std::string
171_encode_impl (T const &data_view)
172{
173 LUMEX_STATIC_ASSERT_MSG (
174 lumex::core::utility::traits::range::has_convertible_size<T>::value,
175 "Encoding data requires a type with a .size() method whose "
176 "return type is convertible to std::size_t.");
177 LUMEX_STATIC_ASSERT_MSG (
178 lumex::core::utility::traits::range::has_convertible_indexed_access<
179 T, Types::byte_type>::value,
180 "Encoding data requires a type with an operator[] that "
181 "accepts std::size_t as an index and returns "
182 "a type convertible to byte_type.");
183
184 if (data_view.empty ())
185 return {};
186
187 // Calculate output length and reserve space
188 std::string result;
189 std::size_t out_length{ 4 * ((data_view.size () + 2) / 3) };
190 result.reserve (out_length);
191
192 // 1. Process 3-byte chunks
193 std::size_t pos{};
194 while (pos + 2 < data_view.size ())
195 {
196 // 1.1. Get the first 6 bits of the first byte
197 result += detail::_base64_chars.at ((data_view[pos] >> 2)
198 & Constants::kBase64MaskSixBits);
199
200 // 1.2. Get the last 2 bits of the first byte and the first 4 bits of the
201 // second byte
202 result += detail::_base64_chars.at (
203 ((data_view[pos] & Constants::kBase64MaskTwoBits) << 4)
204 | ((data_view[pos + 1] >> 4) & Constants::kBase64MaskFourBits));
205
206 // 1.3. Get the last 4 bits of the second byte and the first 2 bits of
207 // the third byte
208 result += detail::_base64_chars.at (
209 ((data_view[pos + 1] & Constants::kBase64MaskFourBits) << 2)
210 | ((data_view[pos + 2] >> Constants::kBase64RightShiftSixBits)
211 & Constants::kBase64MaskTwoBits));
212
213 // 1.4. Get the last 6 bits of the third byte
214 result += detail::_base64_chars.at (data_view[pos + 2]
215 & Constants::kBase64MaskSixBits);
216
217 // 1.5. Move to the next 3 bytes to process
218 pos += 3;
219 }
220
221 // 2. Handle remaining bytes
222 if (pos < data_view.size ())
223 {
224 // 2.1. Get the first 6 bits of the first byte
225 result += detail::_base64_chars.at ((data_view[pos] >> 2)
226 & Constants::kBase64MaskSixBits);
227
228 if (pos + 1 < data_view.size ())
229 {
230 // 2.1.1. Get the last 2 bits of the first byte and the first 4 bits
231 // of the second byte
232 result += detail::_base64_chars.at (
233 ((data_view[pos] & Constants::kBase64MaskTwoBits) << 4)
234 | ((data_view[pos + 1] >> 4) & Constants::kBase64MaskFourBits));
235
236 // 2.1.2. Get the last 4 bits of the second byte
237 result += detail::_base64_chars.at (
238 (data_view[pos + 1] & Constants::kBase64MaskFourBits) << 2);
239
240 // 2.1.3. Add padding
241 result += '=';
242 }
243 else
244 {
245 // 2.2.1. Get the last 2 bits of the first byte and the first 4 bits
246 // of the second byte
247 result += detail::_base64_chars.at (
248 (data_view[pos] & Constants::kBase64MaskTwoBits) << 4);
249
250 // 2.2.2. Add double padding, because there is only one byte left
251 result += "==";
252 }
253 }
254
255 // 3. Return the result
256 return result;
257}
