357 inline std::string
358 logger_format_function_name_clean (char const *prettyFunc) LUMEX_NOEXCEPT
359 {
360  if (prettyFunc == nullptr)
361  return "";
362 
363  std::string func (prettyFunc);
364  if (func.empty ())
365  return "";
366 
367  // Drop arguments (everything after the first '(')
368  std::size_t openParen = func.find ('(');
369  if (openParen != std::string::npos)
370  func = func.substr (0, openParen);
371 
372  // Last space ends the return type (and, on MSVC, the calling convention):
373  // MSVC "bool __cdecl Class::method" / "int __cdecl compute", GCC "void
374  // Class::method" / "void compute". Searching the whole string (not just up
375  // to the last "::") also fixes free functions (no "::" at all), which
376  // previously short-circuited and returned the return type too.
377  std::size_t lastSpace = func.rfind (' ');
378  if (lastSpace != std::string::npos)
379  func = func.substr (lastSpace + 1);
380 
381  return func + "()";
382 }
