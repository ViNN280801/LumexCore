335 inline std::string
336 logger_format_function_name_clean (char const *prettyFunc) LUMEX_NOEXCEPT
337 {
338  if (prettyFunc == nullptr)
339  return "";
340 
341  std::string func (prettyFunc);
342  if (func.empty ())
343  return "";
344 
345  // Drop arguments (everything after the first '(')
346  std::size_t openParen = func.find ('(');
347  if (openParen != std::string::npos)
348  func = func.substr (0, openParen);
349 
350  // Last space ends the return type (and, on MSVC, the calling convention):
351  // MSVC "bool __cdecl Class::method" / "int __cdecl compute", GCC "void
352  // Class::method" / "void compute". Searching the whole string (not just up
353  // to the last "::") also fixes free functions (no "::" at all), which
354  // previously short-circuited and returned the return type too.
355  std::size_t lastSpace = func.rfind (' ');
356  if (lastSpace != std::string::npos)
357  func = func.substr (lastSpace + 1);
358 
359  return func + "()";
360 }
