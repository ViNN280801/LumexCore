315inline std::string
316logger_format_function_name_clean (char const *prettyFunc) LUMEX_NOEXCEPT
317{
318 if (prettyFunc == nullptr)
319 return "";
320
321 std::string func (prettyFunc);
322 if (func.empty ())
323 return "";
324
325 // Drop arguments (everything after the first '(')
326 std::size_t openParen = func.find ('(');
327 if (openParen != std::string::npos)
328 func = func.substr (0, openParen);
329
330 // Last space ends the return type (and, on MSVC, the calling convention):
331 // MSVC "bool __cdecl Class::method" / "int __cdecl compute", GCC "void
332 // Class::method" / "void compute". Searching the whole string (not just up
333 // to the last "::") also fixes free functions (no "::" at all), which
334 // previously short-circuited and returned the return type too.
335 std::size_t lastSpace = func.rfind (' ');
336 if (lastSpace != std::string::npos)
337 func = func.substr (lastSpace + 1);
338
339 return func + "()";
340}
