111template <std::bidirectional_iterator IteratorType,
112 std::sentinel_for<IteratorType> Sentinel, typename ValueType,
113 typename Projection = std::identity,
114 std::indirect_strict_weak_order<
115 ValueType const *, std::projected<IteratorType, Projection>>
116 Predicate
117 = std::ranges::less>
118 requires Detail::NumericConcept<ValueType>
119LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
120LUMEX_CONSTEXPR IteratorType
121 GetNearestTo (IteratorType first, Sentinel last, ValueType const &value,
122 Predicate pred = {}, Projection proj = {})
123{
124 if (first == last)
125 return last;
126
127 auto iter = std::ranges::lower_bound (first, last, value, pred, proj);
128 if (iter == last)
129 return std::ranges::prev (iter);
130 if (iter == first)
131 return iter;
132
133 auto const prev = std::ranges::prev (iter);
134
135 auto const distBetweenFoundAndSpecified
136 = Detail::distance (std::invoke (proj, *iter), value);
137 auto const distBetweenPrevAndSpecified
138 = Detail::distance (std::invoke (proj, *prev), value);
139
140 return (distBetweenFoundAndSpecified < distBetweenPrevAndSpecified) ? iter
141 : prev;
142}

147template <std::ranges::bidirectional_range RangeType, typename ValueType,
148 typename Projection = std::identity,
149 std::indirect_strict_weak_order<
150 ValueType const *,
151 std::projected<std::ranges::iterator_t<RangeType>, Projection>>
152 Predicate
153 = std::ranges::less>
154 requires Detail::NumericConcept<ValueType>
155LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
156LUMEX_CONSTEXPR std::ranges::borrowed_iterator_t<RangeType> GetNearestTo (
157 RangeType &&range, ValueType const &value, Predicate pred = {},
158 Projection proj = {})
159{
160 return GetNearestTo (std::ranges::begin (std::forward<RangeType> (range)),
161 std::ranges::end (std::forward<RangeType> (range)),
162 value, pred, proj);
163}
