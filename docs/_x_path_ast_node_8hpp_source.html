615 template <class T>
616 void
617 step_fill(XPathNodeSetRaw &nsr, // NOLINT(readability-function-cognitive-complexity)
618 XmlNodeBase *node, XPathAllocator *alloc, bool once, T /*val*/)
619 {
620 axis_t const axis = T::axis;
621
622 switch(axis)
623 {
624 case axis_attribute: {
625 for(XmlAttributeBase *attr = node->first_attribute; attr; attr = attr->next_attribute)
626 if(step_push(nsr, attr, node, alloc) & once) return;
627
628 break;
629 }
630
631 case axis_child: {
632 for(XmlNodeBase *ctx = node->first_child; ctx; ctx = ctx->next_sibling)
633 if(step_push(nsr, ctx, alloc) & once) return;
634
635 break;
636 }
637
638 case axis_descendant:
639 case axis_descendant_or_self: {
640 if(axis == axis_descendant_or_self)
641 if(step_push(nsr, node, alloc) & once) return;
642
643 XmlNodeBase *cur = node->first_child;
644
645 while(cur)
646 {
647 if(step_push(nsr, cur, alloc) & once) return;
648
649 if(cur->first_child)
650 cur = cur->first_child;
651 else
652 {
653 while(!cur->next_sibling)
654 {
655 cur = cur->parent;
656
657 if(cur == node) return;
658 }
659
660 cur = cur->next_sibling;
661 }
662 }
663
664 break;
665 }
666
667 case axis_following_sibling: {
668 for(XmlNodeBase *ctx = node->next_sibling; ctx; ctx = ctx->next_sibling)
669 if(step_push(nsr, ctx, alloc) & once) return;
670
671 break;
672 }
673
674 case axis_preceding_sibling: {
675 for(XmlNodeBase *ctx = node->prev_sibling_c; ctx->next_sibling; ctx = ctx->prev_sibling_c)
676 if(step_push(nsr, ctx, alloc) & once) return;
677
678 break;
679 }
680
681 case axis_following: {
682 XmlNodeBase *cur = node;
683
684 // exit from this node so that we don't include descendants
685 while(!cur->next_sibling)
686 {
687 cur = cur->parent;
688
689 if(!cur) return;
690 }
691
692 cur = cur->next_sibling;
693
694 while(cur)
695 {
696 if(step_push(nsr, cur, alloc) & once) return;
697
698 if(cur->first_child)
699 cur = cur->first_child;
700 else
701 {
702 while(!cur->next_sibling)
703 {
704 cur = cur->parent;
705
706 if(!cur) return;
707 }
708
709 cur = cur->next_sibling;
710 }
711 }
712
713 break;
714 }
715
716 case axis_preceding: {
717 XmlNodeBase *cur = node;
718
719 // exit from this node so that we don't include descendants
720 while(!cur->prev_sibling_c->next_sibling)
721 {
722 cur = cur->parent;
723
724 if(!cur) return;
725 }
726
727 cur = cur->prev_sibling_c;
728
729 while(cur)
730 {
731 if(cur->first_child)
732 cur = cur->first_child->prev_sibling_c;
733 else
734 {
735 // leaf node, can't be ancestor
736 if(step_push(nsr, cur, alloc) & once) return;
737
738 while(!cur->prev_sibling_c->next_sibling)
739 {
740 cur = cur->parent;
741
742 if(!cur) return;
743
744 if(!node_is_ancestor(cur, node))
745 if(step_push(nsr, cur, alloc) & once) return;
746 }
747
748 cur = cur->prev_sibling_c;
749 }
750 }
751
752 break;
753 }
754
755 case axis_ancestor:
756 case axis_ancestor_or_self: {
757 if(axis == axis_ancestor_or_self)
758 if(step_push(nsr, node, alloc) & once) return;
759
760 XmlNodeBase *cur = node->parent;
761
762 while(cur)
763 {
764 if(step_push(nsr, cur, alloc) & once) return;
765
766 cur = cur->parent;
767 }
768
769 break;
770 }
771
772 case axis_self: {
773 step_push(nsr, node, alloc);
774
775 break;
776 }
777
778 case axis_parent: {
779 if(node->parent) step_push(nsr, node->parent, alloc);
780
781 break;
782 }
783
784 default: LUMEX_ASSERT(false && "Unimplemented axis"); // unreachable
785 }
786 }

805 template <class T>
806 void
807 step_fill(XPathNodeSetRaw &nsr, // NOLINT(readability-function-cognitive-complexity)
808 XmlAttributeBase *attr, XmlNodeBase *node, XPathAllocator *alloc, bool once, T val)
809 {
810 axis_t const axis = T::axis;
811
812 switch(axis)
813 {
814 case axis_ancestor:
815 case axis_ancestor_or_self: {
816 if(axis == axis_ancestor_or_self
817 && m_test == nodetest_type_node) // reject attributes based on principal node type test
818 if(step_push(nsr, attr, node, alloc) & once) return;
819
820 XmlNodeBase *cur = node;
821
822 while(cur)
823 {
824 if(step_push(nsr, cur, alloc) & once) return;
825
826 cur = cur->parent;
827 }
828
829 break;
830 }
831
832 case axis_descendant_or_self:
833 case axis_self: {
834 if(m_test == nodetest_type_node) // reject attributes based on principal node type test
835 step_push(nsr, attr, node, alloc);
836
837 break;
838 }
839
840 case axis_following: {
841 XmlNodeBase *cur = node;
842
843 while(cur)
844 {
845 if(cur->first_child)
846 cur = cur->first_child;
847 else
848 {
849 while(!cur->next_sibling)
850 {
851 cur = cur->parent;
852
853 if(!cur) return;
854 }
855
856 cur = cur->next_sibling;
857 }
858
859 if(step_push(nsr, cur, alloc) & once) return;
860 }
861
862 break;
863 }
864
865 case axis_parent: {
866 step_push(nsr, node, alloc);
867
868 break;
869 }
870
871 case axis_preceding: {
872 // preceding:: axis does not include attribute nodes and attribute ancestors (they are the same as
873 // parent's ancestors), so we can reuse node preceding
874 step_fill(nsr, node, alloc, once, val);
875 break;
876 }
877
878 default: LUMEX_ASSERT(false && "Unimplemented axis"); // unreachable
879 }

922 template <class T>
923 XPathNodeSetRaw
924 step_do(XPathContext const &ctx, XPathStack const &stack, Types::nodeset_eval_t eval, T val)
925 {
926 axis_t const axis = T::axis;
927 bool const axis_reverse = (axis == axis_ancestor || axis == axis_ancestor_or_self || axis == axis_preceding
928 || axis == axis_preceding_sibling);
929 XPathNodeSet::type_t const axis_type
930 = axis_reverse ? XPathNodeSet::type_sorted_reverse : XPathNodeSet::type_sorted;
931
932 bool once = (axis == axis_attribute && m_test == nodetest_name) || (!m_right && eval_once(axis_type, eval))
933 ||
934 // coverity[mixed_enums]
935 (m_right && !m_right->m_next && m_right->m_test == predicate_constant_one);
936
937 XPathNodeSetRaw nsr;
938 nsr.set_type(axis_type);
939
940 if(m_left)
941 {
942 XPathNodeSetRaw tmp = m_left->eval_node_set(ctx, stack, nodeset_eval_all);
943
944 // self axis preserves the original order
945 if(axis == axis_self) nsr.set_type(tmp.type());
946
947 for(XPathNode const *it = tmp.begin(); it != tmp.end(); ++it)
948 {
949 size_t size = nsr.size();
950
951 // in general, all axes generate elements in a particular order, but there is no order guarantee if axis
952 // is applied to two nodes
953 if(axis != axis_self && size != 0) nsr.set_type(XPathNodeSet::type_unsorted);
954
955 step_fill(nsr, *it, stack.result, once, val);
956 if(m_right) apply_predicates(nsr, size, stack, eval);
957 }
958 }
959 else
960 {
961 step_fill(nsr, ctx.node, stack.result, once, val);
962 if(m_right) apply_predicates(nsr, 0, stack, eval);
963 }
964
965 // child, attribute and self axes always generate unique set of nodes
966 // for other axis, if the set stayed sorted, it stayed unique because the traversal algorithms do not visit
967 // the same node twice
968 if(axis != axis_child && axis != axis_attribute && axis != axis_self
969 && nsr.type() == XPathNodeSet::type_unsorted)
970 nsr.remove_duplicates(stack.temp);
971
972 return nsr;
973 }
