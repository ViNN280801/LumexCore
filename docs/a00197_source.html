859template <typename Allocator>
860std::string
861to_string (LumexBasicStacktrace<Allocator> const &stacktrace)
862{
863 std::string result;
864 result.reserve (stacktrace.size ()
865 * detail::kDefaultMaxFrames); // Pre-allocate for efficiency
866
867 for (std::size_t i = 0; i < stacktrace.size (); ++i)
868 {
869 result += std::to_string (i);
870 result += "# ";
871 result += stacktrace[i].description ();
872 result += '\n';
873 }
874
875 return result;
876}

910template <typename Allocator> struct hash<LumexBasicStacktrace<Allocator>>
911{
918 std::size_t
919 operator() (LumexBasicStacktrace<Allocator> const &stacktrace) const
920 LUMEX_NOEXCEPT
921 {
922 std::size_t seed = 0;
923 for (auto const &entry : stacktrace)
924 {
925 seed ^= std::hash<void *> () (entry.native_handle ())
926 + detail::kHashGoldenRatio + (seed << detail::kHashLeftShift)
927 + (seed >> detail::kHashRightShift);
928 }
929 return seed;
930 }
