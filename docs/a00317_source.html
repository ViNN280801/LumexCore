50// --- dynamic_cast operand forms (pointer / lvalue reference to class) --- //
51template <typename T>
52concept DynCastForm
53 = traits::meta::PointerToClass<T> || traits::meta::LvalueRefToClass<T>;

55template <typename Base, typename Derived>
56concept ValidDownCast
57 = DynCastForm<Base> && DynCastForm<Derived>
58 && (std::is_pointer_v<Base>
59 == std::is_pointer_v<Derived>) // matching ptr "shape"
60 &&traits::meta::CompleteType<traits::meta::indirection_of_t<Base>>
61 && traits::meta::CompleteType<traits::meta::indirection_of_t<Derived>> &&
62 // p5: v must be pointer-to / glvalue-of a polymorphic type
63 std::is_polymorphic_v<
64 std::remove_cv_t<traits::meta::indirection_of_t<Base>>>
65 &&
66 // narrowed to an actual downcast - not an arbitrary cross-cast
67 std::derived_from<
68 std::remove_cv_t<traits::meta::indirection_of_t<Derived>>,
69 std::remove_cv_t<traits::meta::indirection_of_t<Base>>>
70 // p2
71 && traits::meta::PreserveCV<traits::meta::indirection_of_t<Base>,
72 traits::meta::indirection_of_t<Derived>>;

125template <typename Derived, typename Base>
126 requires std::is_pointer_v<Derived> && Detail::ValidDownCast<Base, Derived>
127LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
128Derived downcast (Base base)
129{
130 if (base == nullptr)
131 return nullptr; // p6: null in, null out - not a cast failure.
132
133 Derived result = dynamic_cast<Derived> (base);
134 if (result == nullptr)
135 throw BadDownCast (typeid (*base),
136 typeid (std::remove_pointer_t<Derived>));
137 return result;
138}
