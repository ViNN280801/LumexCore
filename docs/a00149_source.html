1336 void
1337 swap (CircularBuffer &other) LUMEX_NOEXCEPT_IF (is_swap_noexcept)
1338 {
1339 if (std::allocator_traits<Allocator>::propagate_on_container_swap::value)
1340 {
1341 // Swapping allocators (and rebind) is allowed and correct:
1342 std::swap (m_allocator, other.m_allocator);
1343 std::swap (m_slots_alloc, other.m_slots_alloc);
1344 }
1345 else
1346 {
1347 // Allocators do not propagate. Buffer swap requires equivalent
1348 // allocators. Release trusts the documented precondition; debug checks
1349 // it.
1350#ifndef NDEBUG
1351 // Allocator requirements (C++11): equality comparison is allowed.
1352 LUMEX_ASSERT (m_allocator == other.m_allocator);
1353#endif
1354 // Allocators stay in place.
1355 }

1358 std::swap (m_slots, other.m_slots);
1359 std::swap (m_capacity, other.m_capacity);
1360 std::swap (m_size, other.m_size);
1361 std::swap (m_head, other.m_head);
