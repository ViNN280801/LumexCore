1308 template <typename Predicate>

1324 // returned true.
1337 template <typename Predicate>
1338 XmlNode
1339 find_child (Predicate pred) const

1342 return {};

1351 // Find node from subtree using predicate. Returns first node from subtree
1352 // (depth-first), for which predicate returned true.
1368 template <typename Predicate>
1369 XmlNode
1370 find_node (Predicate pred) const
1371 {
1372 if (!m_root)
1373 return {};
1374
1375 XmlNode cur = first_child ();
1376
1377 while (cur.m_root && cur.m_root != m_root)
1378 {
1379 if (pred (cur))
1380 return cur;
1381
1382 if (cur.first_child ())
1383 cur = cur.first_child ();
1384 else if (cur.next_sibling ())
1385 cur = cur.next_sibling ();
1386 else
1387 {
1388 while (!cur.next_sibling () && cur.m_root != m_root)
1389 cur = cur.parent ();
1390
1391 if (cur.m_root != m_root)
1392 cur = cur.next_sibling ();
1393 }
1394 }
1395
1396 return {};
1397 }
