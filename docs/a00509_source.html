269 inline void
270 AppendCrcLeastSignificantByteFirst (crc_params_t const &params,
271  std::vector<std::uint8_t> &buffer)
272  LUMEX_NOEXCEPT
273 {
274  std::uint64_t const value
275  = ComputeCrcWithRevEngParams (params, buffer.data (), buffer.size ());
276  int const numBytes = (params.widthBits + 7) / 8;
277  for (int i = 0; i < numBytes; ++i)
278  buffer.push_back (static_cast<std::uint8_t> ((value >> (8 * i)) & 0xFFU));
279 }
