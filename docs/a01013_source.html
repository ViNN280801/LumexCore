166 {
167  using type = uint16_t;
168 
169  template <typename Traits>
170  static typename Traits::value_type
171  process (uint16_t const *data, std::size_t size,
172  typename Traits::value_type result, Traits /*unused*/)
173  {
174  while (size)
175  {
176  uint16_t lead = opt_swap::value ? endian_swap (*data) : *data;
177 
178  // U+0000..U+D7FF
179  if (lead < 0xD800)
180  { // NOLINT(bugprone-branch-clone)
181  result = Traits::low (result, lead);
182  data += 1;
183  size -= 1;
184  }
185  // U+E000..U+FFFF

189  data += 1;

191  }

201  result,
202  static_cast<uint32_t> (0x10000 + ((lead & 0x3ff) << 10)
203  + (next & 0x3ff)));
204  data += 2;
205  size -= 2;
206  }
207  else
208  {
209  data += 1;
210  size -= 1;
211  }
212  }
213  else
214  {
215  data += 1;
216  size -= 1;
217  }
218  }
219 
220  return result;
221  }

487 struct utf8_decoder
488 {
489  using type = uint8_t;
490 
491  template <typename Traits>
492  static typename Traits::value_type
493  process (uint8_t const *data, std::size_t size,
494  typename Traits::value_type result, Traits /* unused */)
495  {
496  uint8_t const utf8_byte_mask = 0x3f;
497 
498  while (size)
499  {
500  uint8_t lead = *data;
501 
502  // 0xxxxxxx -> U+0000..U+007F
503  if (lead < 0x80)
504  {
505  result = Traits::low (result, lead);
506  data += 1;
507  size -= 1;
508 
509  // process aligned single-byte (ascii) blocks
510  if ((reinterpret_cast<uintptr_t> (data) & 3) == 0)
511  {
512  // round-trip through void* to silence 'cast increases required
513  // alignment of target type' warnings
514  while (
515  size >= 4
516  && (*static_cast<uint32_t const *> (
517  static_cast< // NOLINT(bugprone-casting-through-void)
518  void const *> (data))
519  & 0x80808080)
520  == 0)
521  {
522  result = Traits::low (result, data[0]);
523  result = Traits::low (result, data[1]);
524  result = Traits::low (result, data[2]);
525  result = Traits::low (result, data[3]);
526  data += 4;
527  size -= 4;
528  }
529  }
530  }
531  // 110xxxxx -> U+0080..U+07FF
532  else if (static_cast<unsigned int> (lead - 0xC0) < 0x20 && size >= 2
533  && (data[1] & 0xc0) == 0x80)
534  {
535  result = Traits::low (
536  result, static_cast<uint32_t> (((lead & ~0xC0) << 6)
537  | (data[1] & utf8_byte_mask)));
538  data += 2;
539  size -= 2;
540  }
541  // 1110xxxx -> U+0800-U+FFFF
542  else if (static_cast<unsigned int> (lead - 0xE0) < 0x10 && size >= 3
543  && (data[1] & 0xc0) == 0x80 && (data[2] & 0xc0) == 0x80)
544  {
545  result = Traits::low (
546  result,
547  static_cast<uint32_t> (((lead & ~0xE0) << 12)
548  | ((data[1] & utf8_byte_mask) << 6)
549  | (data[2] & utf8_byte_mask)));
550  data += 3;
551  size -= 3;
552  }
553  // 11110xxx -> U+10000..U+10FFFF
554  else if (static_cast<unsigned int> (lead - 0xF0) < 0x08 && size >= 4
555  && (data[1] & 0xc0) == 0x80 && (data[2] & 0xc0) == 0x80
556  && (data[3] & 0xc0) == 0x80)
557  {
558  result = Traits::high (
559  result,
560  static_cast<uint32_t> (((lead & ~0xF0) << 18)
561  | ((data[1] & utf8_byte_mask) << 12)
562  | ((data[2] & utf8_byte_mask) << 6)
563  | (data[3] & utf8_byte_mask)));
564  data += 4;
565  size -= 4;
566  }
567  // 10xxxxxx or 11111xxx -> invalid
568  else
569  {
570  data += 1;
571  size -= 1;
572  }
573  }
574 
575  return result;
576  }

578 template <typename U>
579 inline U
580 string_to_integer (
581  char_t const *value, // NOLINT(readability-function-cognitive-complexity)
582  U minv, U maxv) // NOLINT(bugprone-easily-swappable-parameters)
583 {
584  U result = 0;
585  char_t const *str = value;
586 
587  while (LUMEX_XML_IS_CHARTYPE (
588  *str,
589  ct_space)) // NOLINT(cppcoreguidelines-pro-bounds-constant-array-index)
590  str++;
591 
592  bool negative = (*str == '-');
593 
594  str += (*str == '+' || *str == '-');
595 
596  bool overflow = false;
597 
598  if (str[0] == '0' && (str[1] | ' ') == 'x')
599  {
600  str += 2;
601 
602  // since overflow detection relies on length of the sequence skip leading
603  // zeros
604  while (*str == '0')
605  str++;
606 
607  char_t const *start = str;
608 
609  for (;;)
610  {
611  if (static_cast<unsigned> (*str - '0') < kDecimalBase)
612  result = (result * kHexadecimalBase) + static_cast<U> (*str - '0');
613  else if (static_cast<unsigned> ((*str | ' ') - 'a')
614  < kHexCharOffsetLimit)
615  result = (result * kHexadecimalBase)
616  + (static_cast<unsigned> ((*str | ' ') - 'a')
617  + kDecimalBase);
618  else
619  break;
620 
621  str++;
622  }
623 
624  auto digits = static_cast<std::size_t> (str - start);
625  overflow = digits > sizeof (U) * 2;
626  }
627  else
628  {
629  // since overflow detection relies on length of the sequence skip leading
630  // zeros
631  while (*str == '0')
632  str++;
633 
634  char_t const *start = str;
635 
636  for (;;)
637  {
638  if (static_cast<unsigned> (*str - '0') < kDecimalBase)
639  result = (result * kDecimalBase) + static_cast<U> (*str - '0');
640  else
641  break;
642 
643  str++;
644  }
645 
646  auto digits = static_cast<std::size_t> (str - start);
647 
648  LUMEX_STATIC_ASSERT (sizeof (U) == kSizeOfU8Bytes || sizeof (U) == 4
649  || sizeof (U) == 2);
650 
651  std::size_t const max_digits10
652  = sizeof (U) == 8 ? 20
653  : sizeof (
654  U) // NOLINT(readability-avoid-nested-conditional-operator)
655  == 4
656  ? 10
657  : 5;
658  char_t const max_lead
659  = sizeof (U) == 8 ? '1'
660  : sizeof (
661  U) // NOLINT(readability-avoid-nested-conditional-operator)
662  == 4
663  ? '4'
664  : '6';
665  std::size_t const high_bit = (sizeof (U) * 8) - 1;
666 
667  overflow = digits >= max_digits10
668  && !(digits == max_digits10
669  && (*start < max_lead
670  || (*start == max_lead && result >> high_bit)));
671  }
672 
673  if (negative)
674  return (overflow || result > 0 - minv) ? minv : 0 - result;
675  return (overflow || result > maxv) ? maxv : result;
676 }

1015 inline bool
1016 hash_insert (void const **table, std::size_t size, void const *key)
1017 {
1018  LUMEX_ASSERT (key);
1019 
1020  auto hash_value
1021  = static_cast<unsigned int> (reinterpret_cast<uintptr_t> (key));
1022 
1023  LUMEX_CONSTEXPR unsigned int kMurmurHash3Finalizer = 0x85ebca6bU;
1024  LUMEX_CONSTEXPR unsigned int kMurmurHash3Multiplier = 0xc2b2ae35U;
1025  LUMEX_CONSTEXPR unsigned int kMurmurHash3Shift16 = 16;
1026  LUMEX_CONSTEXPR unsigned int kMurmurHash3Shift13 = 13;
1027 
1028  // MurmurHash3 32-bit finalizer
1029  hash_value ^= hash_value >> kMurmurHash3Shift16;
1030  hash_value *= kMurmurHash3Finalizer;
1031  hash_value ^= hash_value >> kMurmurHash3Shift13;
1032  hash_value *= kMurmurHash3Multiplier;
1033  hash_value ^= hash_value >> kMurmurHash3Shift16;
1034 
1035  std::size_t hashmod = size - 1;
1036  std::size_t bucket = hash_value & hashmod;
1037 
1038  for (std::size_t probe = 0; probe <= hashmod; ++probe)
1039  {
1040  if (table[bucket] == nullptr)
1041  {
1042  table[bucket] = key;
1043  return true;
1044  }
1045 
1046  if (table[bucket] == key)
1047  return false;
1048 
1049  // hash collision, quadratic probing
1050  bucket = (bucket + probe + 1) & hashmod;
1051  }
1052 
1053  LUMEX_ASSERT (false && "Hash table is full"); // unreachable
1054  return false;
1055 }

1179 inline xml_encoding
1180 guess_buffer_encoding (
1181  uint8_t const *data,
1182  std::size_t size) // NOLINT(readability-function-cognitive-complexity)
1183 {
1184  // skip encoding autodetection if input buffer is too small
1185  if (size < 4)
1186  return encoding_utf8;
1187 
1188  uint8_t d0_ = data[0];
1189  uint8_t d1_ = data[1];
1190  uint8_t d2_ = data[2];
1191  uint8_t d3_ = data[3];
1192 
1193  // look for BOM in first few bytes
1194  if (d0_ == 0 && d1_ == 0 && d2_ == 0xfe && d3_ == 0xff)
1195  return encoding_utf32_be;
1196  if (d0_ == 0xff && d1_ == 0xfe && d2_ == 0 && d3_ == 0)
1197  return encoding_utf32_le;
1198  if (d0_ == 0xfe && d1_ == 0xff)
1199  return encoding_utf16_be;
1200  if (d0_ == 0xff && d1_ == 0xfe)
1201  return encoding_utf16_le;
1202  if (d0_ == 0xef && d1_ == 0xbb && d2_ == 0xbf)
1203  return encoding_utf8;
1204 
1205  // look for <, <? or <?xm in various encodings
1206  if (d0_ == 0 && d1_ == 0 && d2_ == 0 && d3_ == 0x3c)
1207  return encoding_utf32_be;
1208  if (d0_ == 0x3c && d1_ == 0 && d2_ == 0 && d3_ == 0)
1209  return encoding_utf32_le;
1210  if (d0_ == 0 && d1_ == 0x3c && d2_ == 0 && d3_ == 0x3f)
1211  return encoding_utf16_be;
1212  if (d0_ == 0x3c && d1_ == 0 && d2_ == 0x3f && d3_ == 0)
1213  return encoding_utf16_le;
1214 
1215  // look for utf16 < followed by node name (this may fail, but is better than
1216  // utf8 since it's zero terminated so early)
1217  if (d0_ == 0 && d1_ == 0x3c)
1218  return encoding_utf16_be;
1219  if (d0_ == 0x3c && d1_ == 0)
1220  return encoding_utf16_le;
1221 
1222  // no known BOM detected; parse declaration
1223  uint8_t const *enc = nullptr;
1224  std::size_t enc_length = 0;
1225 
1226  if (d0_ == 0x3c && d1_ == 0x3f && d2_ == 0x78 && d3_ == 0x6d
1227  && parse_declaration_encoding (data, size, enc, enc_length))
1228  {
1229  // iso-8859-1 (case-insensitive)
1230  if (enc_length == 10 && (enc[0] | ' ') == 'i' && (enc[1] | ' ') == 's'
1231  && (enc[2] | ' ') == 'o' && enc[3] == '-' && enc[4] == '8'
1232  && enc[5] == '8' && enc[6] == '5' && enc[7] == '9' && enc[8] == '-'
1233  && enc[9] == '1')
1234  return encoding_latin1;
1235 
1236  // latin1 (case-insensitive)
1237  if (enc_length == 6 && (enc[0] | ' ') == 'l' && (enc[1] | ' ') == 'a'
1238  && (enc[2] | ' ') == 't' && (enc[3] | ' ') == 'i'
1239  && (enc[4] | ' ') == 'n' && enc[5] == '1')
1240  return encoding_latin1;
1241  }
1242 
1243  return encoding_utf8;
1244 }
