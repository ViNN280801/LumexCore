202 // 1 option. Define the exception class: LUMEX_DEFINE_EXCEPTION and
203 // LUMEX_DEFINE_EXCEPTION_WITH_BODY live in LumexExceptionMacros.hpp (included
204 // above) so modules that must not depend on this one can use them too.
205 
206 // 2 option. Throw the exception.
207 // Pattern:
208 // [exception_name] -> custom message
209 // Example:
210 // [LumexException] -> Failed to open file
211 #include "lumex/core/string/LumexString"
212 #define LUMEX_THROW_EXCEPTION(exception_name, msg) \
213  throw exception_name (lumex::core::string::utility::stringify ( \
214  lumDemangle (exception_name), ": ", msg));
215 
216 // 3. Handle the exception.
217 #define LUMEX_EXCEPTION_HANDLE_BEGIN \
218  try \
219  { \

221 
222 #define LUMEX_EXCEPTION_HANDLE_END \
223  } \
224  catch (LumexBaseException const &ex) \
225  { \
226  ex.to_stderr (); \
227  ex.to_crash_report (); \
228  } \
229  catch (std::exception const &ex) \
230  { \
231  std::cerr << "[std::exception] " << ex.what () << '\n'; \
