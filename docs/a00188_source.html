2463 _setupWindowsHandlers();
2464#elif DUMP_CREATOR_UNIX
2465 _setupSignalHandlers();
2466 _setupCoreDumpSettings();
2467
2468 // Store orig core_pattern BEFORE attempting to change it
2469 _setupCorePattern();
2470 std::string pattern_before_change = s_originalCorePattern;
2471
2472 // Try to set custom core_pattern via sudo (primary mechanism)
2473 _generateCoreDump();
2474
2475 // Verify if core_pattern actually changed with kernel specifiers
2476 std::ifstream check_pattern("/proc/sys/kernel/core_pattern");
2477 std::string pattern_after_change;
2478
2479 if(std::getline(check_pattern, pattern_after_change))
2480 {
2481 // Check for our pattern with kernel specifiers: %t, %p, %e
2482 bool has_kernel_specifiers
2483 = (pattern_after_change.find("%t") != std::string::npos && pattern_after_change.find("%p") != std::string::npos
2484 && pattern_after_change.find("%e") != std::string::npos);
2485
2486 bool has_our_directory = (pattern_after_change.find(s_dumpDirectory) != std::string::npos);
2487
2488 if(has_kernel_specifiers && has_our_directory)
2489 {
2490 _logMessage("SUCCESS: Custom core_pattern set successfully", false);
2491 _logMessage(" Dumps will be created directly in: " + s_dumpDirectory, false);
2492 _logMessage(" Pattern: " + pattern_after_change, false);
2493 }
2494 else
2495 {
2496 _logMessage("WARNING: core_pattern did not change to our pattern (sudo likely failed)", false);
2497
2498 // Try to restore original systemd-coredump pattern if it was active before
2499 // This prevents leaving stale pattern from previous run
2500 if(!pattern_before_change.empty() && pattern_before_change[0] == '|'
2501 && pattern_before_change.find("systemd-coredump") != std::string::npos)
2502 {
2503 _logMessage("INFO: Attempting to restore original systemd-coredump pattern", false);
2504 std::string restore_cmd
2505 = "echo \"" + pattern_before_change + "\" | sudo tee /proc/sys/kernel/core_pattern >/dev/null 2>&1";
2506 int restore_ret = system(restore_cmd.c_str());
2507
2508 if(restore_ret == 0)
2509 {
2510 _logMessage("INFO: Successfully restored systemd-coredump pattern", false);
2511 pattern_after_change = pattern_before_change; // Update for subsequent checks
2512 }
2513 }
2514
