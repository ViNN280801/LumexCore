145inline void
146write_to_stderr (std::string const &prefix, char const *whatMsg) LUMEX_NOEXCEPT
147{
148 std::cerr.write (prefix.data (),
149 static_cast<std::streamsize> (prefix.size ()));
150
151 static char const kReason[] = ". Reason: ";
152 std::cerr.write (kReason,
153 static_cast<std::streamsize> (sizeof (kReason) - 1));
154
155 if (whatMsg != nullptr)
156 {
157 std::cerr.write (whatMsg,
158 static_cast<std::streamsize> (std::strlen (whatMsg)));
159 }
160 else
161 {
162 static char const kUnknownWhat[] = "<unknown exception message>";
163 std::cerr.write (kUnknownWhat, static_cast<std::streamsize> (
164 sizeof (kUnknownWhat) - 1));
165 }
166
167 std::cerr.put ('\n');
168 std::cerr.flush ();
169}

275template <typename Function, typename... Args>
276auto
277ExceptionWrapper (
278 std::string const
279 &excMessage, // NOLINT(bugprone-easily-swappable-parameters)
280 std::string const &unknownExcMessage, Function &&func, Args &&...args)
281 LUMEX_NOEXCEPT_IF (noexcept (
282 std::forward<Function> (func) (std::forward<Args> (args)...)))
283 -> lumex::core::utility::traits::invoke::invoke_result_t<Function,
284 Args...>
285{
286 using ReturnType
287 = lumex::core::utility::traits::invoke::invoke_result_t<Function,
288 Args...>;
289
290 LUMEX_STATIC_ASSERT_MSG (
291 lumex::core::utility::traits::invoke::is_callable_v<Function, Args...>,
292 "Function must be callable with the supplied arguments. Check:\n"
293 "1) Function type is correct\n"
294 "2) Number of arguments matches\n"
295 "3) Argument types are compatible");
296
297 LUMEX_STATIC_ASSERT_MSG (
298 std::is_default_constructible<ReturnType>::value
299 || std::is_void<ReturnType>::value,
300 "ReturnType must be default-constructible or void. "
301 "For non-default-constructible types, consider using std::optional.");
302
303 try
304 {
305 return std::forward<Function> (func) (std::forward<Args> (args)...);
306 }
307 catch (std::exception const &exc)
308 {
309 detail::report_exception (excMessage, exc.what ());
310 }
311 catch (...)
312 {
313 detail::report_line (unknownExcMessage);
314 }
315
316 // static_assert above guarantees ReturnType is default-constructible or
317 // void, so this cannot throw.
318 return lumex::core::utility::traits::meta::default_return<
319 ReturnType>::value ();
320}
