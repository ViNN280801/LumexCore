989 template <typename Char>
990 LUMEX_FORMAT_CONSTEXPR void
991 validate_specs (format_specs_t<Char> const &specs, SpecKind kind)
992 {
993  Char const type = specs.type;
994  bool numeric = false;
995  bool allows_precision = false;
996  bool allows_localized = false;
997  bool allows_zero = false;
998  char const *const invalid = kind == SpecKind::character
999  ? "invalid format specifier for char"
1000  : "invalid format specifier";
1001  switch (kind)
1002  {
1003  case SpecKind::integer:
1004  if (type != Char () && type != static_cast<Char> ('c')
1005  && !is_integer_presentation (type))
1006  throw FormatError ("invalid format specifier");
1007  numeric = type != static_cast<Char> ('c');
1008  allows_localized = numeric;
1009  allows_zero = numeric;
1010  break;
1011  case SpecKind::character:
1012  if (type != Char () && type != static_cast<Char> ('c')
1013  && type != static_cast<Char> ('?')
1014  && !is_integer_presentation (type))
1015  throw FormatError ("invalid format specifier");
1016  numeric = is_integer_presentation (type);
1017  allows_localized = numeric;
1018  allows_zero = numeric;
1019  break;
1020  case SpecKind::boolean:
1021  if (type != Char () && type != static_cast<Char> ('s')
1022  && !is_integer_presentation (type))
1023  throw FormatError ("invalid format specifier");
1024  numeric = is_integer_presentation (type);
1025  allows_localized = true;
1026  allows_zero = numeric;
1027  break;
1028  case SpecKind::floating:
1029  if (type != Char () && !is_float_presentation (type))
1030  throw FormatError ("invalid format specifier");
1031  numeric = true;
1032  allows_precision = true;
1033  allows_localized = true;
1034  allows_zero = true;
1035  break;
1036  case SpecKind::string:
1037  if (type != Char () && type != static_cast<Char> ('s')
1038  && type != static_cast<Char> ('?'))
1039  throw FormatError ("invalid format specifier");
1040  allows_precision = true;
1041  break;
1042  case SpecKind::pointer:
1043  if (type != Char () && type != static_cast<Char> ('p')
1044  && type != static_cast<Char> ('P'))
1045  throw FormatError ("invalid format specifier");
1046  // P2510 (a defect report against C++20): zeros pad after "0x".
1047  allows_zero = true;
1048  break;
1049  case SpecKind::enumeration:
1050  if (type != Char () && type != static_cast<Char> ('s')
1051  && !is_integer_presentation (type))
1052  throw FormatError ("invalid format specifier");
1053  numeric = is_integer_presentation (type);
1054  allows_precision = !numeric;
1055  allows_localized = numeric;
1056  allows_zero = numeric;
1057  break;
1058  }
1059  if ((specs.sign != Sign::none || specs.alternate) && !numeric)
1060  throw FormatError (invalid);
1061  if (specs.zero_pad && !allows_zero)
1062  throw FormatError (kind == SpecKind::character
1063  ? invalid
1064  : "format specifier requires numeric argument");
1065  if ((specs.precision >= 0 || specs.precision_ref.kind != ArgRefKind::none)
1066  && !allows_precision)
1067  throw FormatError ("invalid format specifier");
1068  if (specs.localized && !allows_localized)
1069  throw FormatError ("invalid format specifier");
1070 }

1077 template <typename Char>
1078 LUMEX_FORMAT_CONSTEXPR Char const *
1079 parse_std_specs (BasicFormatParseContext<Char> &ctx,
1080  format_specs_t<Char> &specs, SpecKind kind)
1081 {
1082  Char const *it = ctx.begin ();
1083  Char const *const end = ctx.end ();
1084  if (it == end || *it == static_cast<Char> ('}'))
1085  {
1086  validate_specs (specs, kind);
1087  return it;
1088  }
1089 
1090  // [[fill]align]
1091  std::size_t const fill_length = code_point_length (it, end);
1092  if (static_cast<std::size_t> (end - it) > fill_length
1093  && is_align (it[fill_length]))
1094  {
1095  if (*it == static_cast<Char> ('{'))
1096  throw FormatError ("invalid fill character '{'");
1097  if (*it == static_cast<Char> ('}'))
1098  throw FormatError ("invalid fill character '}'");
1099  for (std::size_t i = 0; i < fill_length; ++i)
1100  specs.fill[i] = it[i];
1101  specs.fill_size = static_cast<unsigned char> (fill_length);
1102  it += fill_length;
1103  }
1104  if (it != end && is_align (*it))
1105  {
1106  specs.align = *it == static_cast<Char> ('<') ? Align::left
1107  : *it == static_cast<Char> ('>') ? Align::right
1108  : Align::center;
1109  ++it;
1110  }
1111 
1112  // [sign]
1113  if (it != end
1114  && (*it == static_cast<Char> ('+') || *it == static_cast<Char> ('-')
1115  || *it == static_cast<Char> (' ')))
1116  {
1117  specs.sign = *it == static_cast<Char> ('+') ? Sign::plus
1118  : *it == static_cast<Char> ('-') ? Sign::minus
1119  : Sign::space;
1120  ++it;
1121  }
1122 
1123  // [#]
1124  if (it != end && *it == static_cast<Char> ('#'))
1125  {
1126  specs.alternate = true;
1127  ++it;
1128  }
1129 
1130  // [0]
1131  if (it != end && *it == static_cast<Char> ('0'))
1132  {
1133  specs.zero_pad = true;
1134  ++it;
1135  // The width that may follow starts with a nonzero digit (`{:00}`
1136  // is an error, as in std::format).
1137  if (it != end && *it == static_cast<Char> ('0'))
1138  throw FormatError ("invalid format specifier");
1139  }
1140 
1141  // [width]
1142  if (it != end && is_digit (*it))
1143  specs.width = parse_nonnegative_int (it, end);
1144  else if (it != end && *it == static_cast<Char> ('{'))
1145  specs.width_ref = parse_dynamic_ref (it, end, ctx);
1146 
1147  // [.precision]
1148  if (it != end && *it == static_cast<Char> ('.'))
1149  {
1150  ++it;
1151  if (it != end && is_digit (*it))
1152  specs.precision = parse_nonnegative_int (it, end);
1153  else if (it != end && *it == static_cast<Char> ('{'))
1154  specs.precision_ref = parse_dynamic_ref (it, end, ctx);
1155  else if (it == end)
1156  throw FormatError ("invalid precision");
1157  else
1158  throw FormatError ("invalid format string");
1159  }
1160 
1161  // [L]
1162  if (it != end && *it == static_cast<Char> ('L'))
1163  {
1164  specs.localized = true;
1165  ++it;
1166  }
1167 
1168  // [type]
1169  if (it != end && *it != static_cast<Char> ('}'))
1170  {
1171  specs.type = *it;
1172  ++it;
1173  }
1174 
1175  if (it != end && *it != static_cast<Char> ('}'))
1176  throw FormatError ("invalid format specifier");
1177  validate_specs (specs, kind);
1178  return it;
1179 }

1206 template <typename Char, typename Handler>
1207 LUMEX_FORMAT_CONSTEXPR void
1208 parse_format_string (Char const *begin, Char const *end, Handler &handler)
1209 {
1210  Char const *text = begin;
1211  Char const *it = begin;
1212  while (it != end)
1213  {
1214  Char const value = *it;
1215  if (value == static_cast<Char> ('{'))
1216  {
1217  handler.on_text (text, it);
1218  handler.on_field_start (static_cast<std::size_t> (it - begin));
1219  ++it;
1220  if (it == end)
1221  throw FormatError ("invalid format string");
1222  if (*it == static_cast<Char> ('{'))
1223  {
1224  handler.on_text (it, it + 1);
1225  ++it;
1226  text = it;
1227  continue;
1228  }
1229  if (*it == static_cast<Char> ('}') || *it == static_cast<Char> (':'))
1230  handler.on_auto_arg ();
1231  else
1232  {
1233  arg_ref_t<Char> const ref = parse_arg_id (it, end);
1234  if (ref.kind == ArgRefKind::index)
1235  handler.on_index_arg (ref.index);
1236  else
1237  handler.on_name_arg (ref.name, ref.name_size);
1238  }
1239  if (it == end)
1240  throw FormatError ("invalid format string");
1241  if (*it == static_cast<Char> (':'))
1242  ++it;
1243  else if (*it != static_cast<Char> ('}'))
1244  throw FormatError ("invalid format string");
1245  it = handler.on_format_specs (it, end);
1246  if (it == end)
1247  throw FormatError ("missing '}' in format string");
1248  if (*it != static_cast<Char> ('}'))
1249  throw FormatError ("unknown format specifier");
1250  ++it;
1251  text = it;
1252  }
1253  else if (value == static_cast<Char> ('}'))
1254  {
1255  handler.on_text (text, it);
1256  handler.on_field_start (static_cast<std::size_t> (it - begin));
1257  ++it;
1258  if (it == end || *it != static_cast<Char> ('}'))
1259  throw FormatError ("unmatched '}' in format string");
1260  handler.on_text (it, it + 1);
1261  ++it;
1262  text = it;
1263  }
1264  else
1265  ++it;
1266  }
1267  handler.on_text (text, end);
1268 }

1541  }

1543  }
