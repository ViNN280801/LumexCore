747template <typename Char>
748LUMEX_FORMAT_CONSTEXPR std::size_t
749code_point_length (Char const *it, Char const *end) LUMEX_NOEXCEPT
750{
751 std::size_t length = 1;
752 if (sizeof (Char) == 1)
753 {
754 unsigned char const lead = static_cast<unsigned char> (*it);
755 if (lead >= 0xF0 && lead < 0xF8)
756 length = 4;
757 else if (lead >= 0xE0)
758 length = lead < 0xF0 ? 3 : 1;
759 else if (lead >= 0xC0)
760 length = 2;
761 }
762 else if (sizeof (Char) == 2)
763 {
764 unsigned const unit = static_cast<unsigned> (*it) & 0xFFFFu;
765 if (unit >= 0xD800u && unit < 0xDC00u)
766 length = 2;
767 }
768 if (static_cast<std::size_t> (end - it) < length)
769 length = 1;
770 return length;

794template <typename Char>
795LUMEX_FORMAT_CONSTEXPR arg_ref_t<Char>
796parse_arg_id (Char const *&it, Char const *end)
797{
798 arg_ref_t<Char> ref;
799 if (it != end && is_digit (*it))
800 {
801 if (*it == static_cast<Char> ('0') && it + 1 != end && is_digit (it[1]))
802 throw FormatError ("invalid format string");
803 // An index above INT_MAX cannot name an argument: saturate, so the
804 // lookup reports "argument not found" rather than a number error.
805 unsigned long long value = 0;
806 while (it != end && is_digit (*it))
807 {
808 if (value <= static_cast<unsigned long long> (INT_MAX))
809 value = value * 10
810 + static_cast<unsigned> (*it - static_cast<Char> ('0'));
811 ++it;
812 }
813 ref.kind = ArgRefKind::index;
814 ref.index = value > static_cast<unsigned long long> (INT_MAX)
815 ? INT_MAX
816 : static_cast<int> (value);
817 return ref;
818 }
819 if (it != end && is_name_start (*it))
820 {
821 Char const *const start = it;
822 while (it != end && (is_name_start (*it) || is_digit (*it)))
823 ++it;
824 ref.kind = ArgRefKind::name;
825 ref.name = start;
826 ref.name_size = static_cast<std::size_t> (it - start);
827 return ref;
828 }
829 throw FormatError ("invalid format string");
830}

998 char const *const invalid = kind == SpecKind::character
999 ? "invalid format specifier for char"
1000 : "invalid format specifier";
1001 switch (kind)
1002 {
1003 case SpecKind::integer:
1004 if (type != Char () && type != static_cast<Char> ('c')
1005 && !is_integer_presentation (type))
1006 throw FormatError ("invalid format specifier");
1007 numeric = type != static_cast<Char> ('c');
1008 allows_localized = numeric;
1009 allows_zero = numeric;
1010 break;
1011 case SpecKind::character:
1012 if (type != Char () && type != static_cast<Char> ('c')
1013 && type != static_cast<Char> ('?')
1014 && !is_integer_presentation (type))
1015 throw FormatError ("invalid format specifier");
1016 numeric = is_integer_presentation (type);
1017 allows_localized = numeric;
1018 allows_zero = numeric;
1019 break;
1020 case SpecKind::boolean:
1021 if (type != Char () && type != static_cast<Char> ('s')
1022 && !is_integer_presentation (type))
1023 throw FormatError ("invalid format specifier");
1024 numeric = is_integer_presentation (type);
1025 allows_localized = true;
1026 allows_zero = numeric;
1027 break;
1028 case SpecKind::floating:
1029 if (type != Char () && !is_float_presentation (type))
1030 throw FormatError ("invalid format specifier");
1031 numeric = true;
1032 allows_precision = true;
1033 allows_localized = true;
1034 allows_zero = true;
1035 break;
1036 case SpecKind::string:
1037 if (type != Char () && type != static_cast<Char> ('s')
1038 && type != static_cast<Char> ('?'))
1039 throw FormatError ("invalid format specifier");
1040 allows_precision = true;
1041 break;
1042 case SpecKind::pointer:
1043 if (type != Char () && type != static_cast<Char> ('p')
1044 && type != static_cast<Char> ('P'))
1045 throw FormatError ("invalid format specifier");
1046 // P2510 (a defect report against C++20): zeros pad after "0x".
1047 allows_zero = true;
1048 break;
1049 case SpecKind::enumeration:
1050 if (type != Char () && type != static_cast<Char> ('s')
1051 && !is_integer_presentation (type))
1052 throw FormatError ("invalid format specifier");
1053 numeric = is_integer_presentation (type);
1054 allows_precision = !numeric;
1055 allows_localized = numeric;
1056 allows_zero = numeric;
1057 break;
1058 }
1059 if ((specs.sign != Sign::none || specs.alternate) && !numeric)
1060 throw FormatError (invalid);
1061 if (specs.zero_pad && !allows_zero)
1062 throw FormatError (kind == SpecKind::character
1063 ? invalid
1064 : "format specifier requires numeric argument");
1065 if ((specs.precision >= 0 || specs.precision_ref.kind != ArgRefKind::none)
1066 && !allows_precision)
1067 throw FormatError ("invalid format specifier");
1068 if (specs.localized && !allows_localized)
1069 throw FormatError ("invalid format specifier");
1070}

1077template <typename Char>
1078LUMEX_FORMAT_CONSTEXPR Char const *
1079parse_std_specs (BasicFormatParseContext<Char> &ctx,
1080 format_specs_t<Char> &specs, SpecKind kind)
1081{
1082 Char const *it = ctx.begin ();
1083 Char const *const end = ctx.end ();
1084 if (it == end || *it == static_cast<Char> ('}'))
1085 {
1086 validate_specs (specs, kind);
1087 return it;
1088 }
1089
1090 // [[fill]align]
1091 std::size_t const fill_length = code_point_length (it, end);
1092 if (static_cast<std::size_t> (end - it) > fill_length
1093 && is_align (it[fill_length]))
1094 {
1095 if (*it == static_cast<Char> ('{'))
1096 throw FormatError ("invalid fill character '{'");
1097 if (*it == static_cast<Char> ('}'))
1098 throw FormatError ("invalid fill character '}'");
1099 for (std::size_t i = 0; i < fill_length; ++i)
1100 specs.fill[i] = it[i];
1101 specs.fill_size = static_cast<unsigned char> (fill_length);
1102 it += fill_length;
1103 }
1104 if (it != end && is_align (*it))
1105 {
1106 specs.align = *it == static_cast<Char> ('<') ? Align::left
1107 : *it == static_cast<Char> ('>') ? Align::right
1108 : Align::center;
1109 ++it;
1110 }
1111
1112 // [sign]
1113 if (it != end
1114 && (*it == static_cast<Char> ('+') || *it == static_cast<Char> ('-')
1115 || *it == static_cast<Char> (' ')))
1116 {
1117 specs.sign = *it == static_cast<Char> ('+') ? Sign::plus
1118 : *it == static_cast<Char> ('-') ? Sign::minus
1119 : Sign::space;
1120 ++it;
1121 }
1122
1123 // [#]
1124 if (it != end && *it == static_cast<Char> ('#'))
1125 {
1126 specs.alternate = true;
1127 ++it;
1128 }
1129
1130 // [0]
1131 if (it != end && *it == static_cast<Char> ('0'))
1132 {
1133 specs.zero_pad = true;
1134 ++it;
1135 // The width that may follow starts with a nonzero digit (`{:00}`
1136 // is an error, as in std::format).
1137 if (it != end && *it == static_cast<Char> ('0'))
1138 throw FormatError ("invalid format specifier");
1139 }
1140
1141 // [width]
1142 if (it != end && is_digit (*it))
1143 specs.width = parse_nonnegative_int (it, end);
1144 else if (it != end && *it == static_cast<Char> ('{'))
1145 specs.width_ref = parse_dynamic_ref (it, end, ctx);
1146
1147 // [.precision]
1148 if (it != end && *it == static_cast<Char> ('.'))
1149 {
1150 ++it;
1151 if (it != end && is_digit (*it))
1152 specs.precision = parse_nonnegative_int (it, end);
1153 else if (it != end && *it == static_cast<Char> ('{'))
1154 specs.precision_ref = parse_dynamic_ref (it, end, ctx);
1155 else if (it == end)
1156 throw FormatError ("invalid precision");
1157 else
1158 throw FormatError ("invalid format string");
1159 }
1160
1161 // [L]
1162 if (it != end && *it == static_cast<Char> ('L'))
1163 {
1164 specs.localized = true;
1165 ++it;
1166 }
1167
1168 // [type]
1169 if (it != end && *it != static_cast<Char> ('}'))
1170 {
1171 specs.type = *it;
1172 ++it;
1173 }
1174
1175 if (it != end && *it != static_cast<Char> ('}'))
1176 throw FormatError ("invalid format specifier");
1177 validate_specs (specs, kind);
1178 return it;
1179}

1182template <typename Char>
1183LUMEX_FORMAT_CONSTEXPR Char const *
1184skip_specs (Char const *it, Char const *end)
1185{
1186 int depth = 0;
1187 for (; it != end; ++it)
1188 {
1189 if (*it == static_cast<Char> ('{'))
1190 ++depth;
1191 else if (*it == static_cast<Char> ('}'))
1192 {
1193 if (depth == 0)
1194 return it;
1195 --depth;
1196 }
1197 }

1206template <typename Char, typename Handler>
1207LUMEX_FORMAT_CONSTEXPR void
1208parse_format_string (Char const *begin, Char const *end, Handler &handler)
1209{
1210 Char const *text = begin;
1211 Char const *it = begin;
1212 while (it != end)
1213 {
1214 Char const value = *it;
1215 if (value == static_cast<Char> ('{'))
1216 {
1217 handler.on_text (text, it);
1218 handler.on_field_start (static_cast<std::size_t> (it - begin));
1219 ++it;
1220 if (it == end)
1221 throw FormatError ("invalid format string");
1222 if (*it == static_cast<Char> ('{'))
1223 {
1224 handler.on_text (it, it + 1);
1225 ++it;
1226 text = it;
1227 continue;
1228 }
1229 if (*it == static_cast<Char> ('}') || *it == static_cast<Char> (':'))
1230 handler.on_auto_arg ();
1231 else
1232 {
1233 arg_ref_t<Char> const ref = parse_arg_id (it, end);
1234 if (ref.kind == ArgRefKind::index)
1235 handler.on_index_arg (ref.index);
1236 else
1237 handler.on_name_arg (ref.name, ref.name_size);
1238 }
1239 if (it == end)
1240 throw FormatError ("invalid format string");
1241 if (*it == static_cast<Char> (':'))
1242 ++it;
1243 else if (*it != static_cast<Char> ('}'))
1244 throw FormatError ("invalid format string");
1245 it = handler.on_format_specs (it, end);
1246 if (it == end)
1247 throw FormatError ("missing '}' in format string");
1248 if (*it != static_cast<Char> ('}'))
1249 throw FormatError ("unknown format specifier");
1250 ++it;
1251 text = it;
1252 }
1253 else if (value == static_cast<Char> ('}'))
1254 {
1255 handler.on_text (text, it);
1256 handler.on_field_start (static_cast<std::size_t> (it - begin));
1257 ++it;
1258 if (it == end || *it != static_cast<Char> ('}'))
1259 throw FormatError ("unmatched '}' in format string");
1260 handler.on_text (it, it + 1);
1261 ++it;
1262 text = it;
1263 }
1264 else
1265 ++it;
1266 }
1267 handler.on_text (text, end);
1268}

1497template <typename Char>
1498std::uint32_t
1499decode (Char const *&it, Char const *end, bool &valid) LUMEX_NOEXCEPT
1500{
1501 valid = true;
1502 if (sizeof (Char) == 1)
1503 {
1504 unsigned char const lead = static_cast<unsigned char> (*it);
1505 std::size_t length = 0;
1506 std::uint32_t cp = 0;
1507 if (lead < 0x80)
1508 {
1509 ++it;
1510 return lead;
1511 }
1512 if (lead >= 0xC2 && lead < 0xE0)
1513 {
1514 length = 2;
1515 cp = lead & 0x1Fu;
1516 }
1517 else if (lead >= 0xE0 && lead < 0xF0)
1518 {
1519 length = 3;
1520 cp = lead & 0x0Fu;
1521 }
1522 else if (lead >= 0xF0 && lead < 0xF5)
1523 {
1524 length = 4;
1525 cp = lead & 0x07u;
1526 }
1527 if (length == 0 || static_cast<std::size_t> (end - it) < length)
1528 {
1529 valid = false;
1530 ++it;
1531 return lead;
1532 }
1533 for (std::size_t i = 1; i < length; ++i)
1534 {
1535 unsigned char const unit = static_cast<unsigned char> (it[i]);
1536 if ((unit & 0xC0u) != 0x80u)
1537 {
1538 valid = false;
1539 ++it;
1540 return lead;
1541 }
1542 cp = (cp << 6) | (unit & 0x3Fu);
1543 }
1544 it += length;
1545 return cp;
1546 }
1547 if (sizeof (Char) == 2)
1548 {
1549 std::uint32_t const unit = static_cast<std::uint32_t> (*it) & 0xFFFFu;
1550 if (unit >= 0xD800u && unit < 0xDC00u && end - it >= 2)
1551 {
1552 std::uint32_t const low
1553 = static_cast<std::uint32_t> (it[1]) & 0xFFFFu;
1554 if (low >= 0xDC00u && low < 0xE000u)
1555 {
1556 it += 2;
1557 return 0x10000u + ((unit - 0xD800u) << 10) + (low - 0xDC00u);
1558 }
1559 }
1560 if (unit >= 0xD800u && unit < 0xE000u)
1561 valid = false;
1562 ++it;
1563 return unit;
1564 }
1565 std::uint32_t const unit = static_cast<std::uint32_t> (*it);
1566 ++it;
1567 return unit;
1568}

1735template <typename Char>
1736std::size_t
1737next_grapheme (Char const *&it, Char const *end) LUMEX_NOEXCEPT
1738{
1739 if (*it == static_cast<Char> ('\r') && it + 1 != end
1740 && it[1] == static_cast<Char> ('\n'))
1741 {
1742 it += 2; // CR LF is one cluster
1743 return 1;
1744 }
1745 if (static_cast<std::uint32_t> (*it) < 0x80u
1746 && (it + 1 == end || static_cast<std::uint32_t> (it[1]) < 0x80u))
1747 {
1748 ++it; // ASCII followed by ASCII (or the end): the common fast path
1749 return 1;
1750 }
1751 bool valid = true;
1752 std::uint32_t const first = decode (it, end, valid);
1753 if (!valid)
1754 return 1;
1755 std::size_t const width = is_wide_code_point (first) ? 2 : 1;
1756 std::uint32_t previous = first;
1757 bool regional_pair_open = is_regional_indicator (first);
1758 while (it != end)
1759 {
1760 Char const *next = it;
1761 bool next_valid = true;
1762 std::uint32_t const cp = decode (next, end, next_valid);
1763 if (!next_valid)
1764 break;
1765 bool joins = is_grapheme_extend (cp) || cp == 0x200Du;
1766 if (!joins && previous == 0x200Du && is_extended_pictographic (cp))
1767 joins = true;
1768 if (!joins && regional_pair_open && is_regional_indicator (cp))
1769 {
1770 joins = true;
1771 regional_pair_open = false;
1772 }
1773 if (!joins)
1774 break;
1775 previous = cp;
1776 it = next;
1777 }
1778 return width;
1779}

1882template <typename Char>
1883std::basic_string<Char>
1884group_digits (std::basic_string<Char> const &digits,
1885 std::string const &grouping, Char separator)
1886{
1887 if (grouping.empty () || digits.empty ())
1888 return digits;
1889 std::basic_string<Char> reversed;
1890 std::size_t group_index = 0;
1891 int group_size = static_cast<unsigned char> (grouping[0]);
1892 int in_group = 0;
1893 for (std::size_t i = digits.size (); i-- > 0;)
1894 {
1895 if (group_size > 0 && group_size < CHAR_MAX && in_group == group_size)
1896 {
1897 reversed.push_back (separator);
1898 in_group = 0;
1899 if (group_index + 1 < grouping.size ())
1900 {
1901 ++group_index;
1902 group_size = static_cast<unsigned char> (grouping[group_index]);
1903 }
1904 }
1905 reversed.push_back (digits[i]);
1906 ++in_group;
1907 }
1908 return std::basic_string<Char> (reversed.rbegin (), reversed.rend ());
1909}

1946template <typename Char, typename UInt>
1947void
1948write_integer (Buffer<Char> &buffer, format_specs_t<Char> const &specs,
1949 bool negative, UInt magnitude,
1950 BasicFormatContext<Char> const &ctx)
1951{
1952 Char const type = specs.type;
1953 if (type == static_cast<Char> ('c'))
1954 {
1955 // The value must fit the character type itself (as std::format):
1956 // -128..127 for a signed `char`, 0..65535 for a 16-bit `wchar_t`.
1957 unsigned long long const max_positive = static_cast<unsigned long long> (
1958 (std::numeric_limits<Char>::max) ());
1959 unsigned long long const max_negative
1960 = std::is_signed<Char>::value
1961 ? 0ull
1962 - static_cast<unsigned long long> (
1963 static_cast<long long> (
1964 (std::numeric_limits<Char>::min) ()))
1965 : 0ull;
1966 if (magnitude
1967 > static_cast<UInt> (negative ? max_negative : max_positive))
1968 throw FormatError ("character out of range");
1969 unsigned long long const bits
1970 = static_cast<unsigned long long> (magnitude);
1971 Char const value
1972 = negative ? static_cast<Char> (0ll - static_cast<long long> (bits))
1973 : static_cast<Char> (bits);
1974 write_character (buffer, specs, value);
1975 return;
1976 }
1977 unsigned base = 10;
1978 std::string prefix;
1979 if (type == static_cast<Char> ('b') || type == static_cast<Char> ('B'))
1980 {
1981 base = 2;
1982 if (specs.alternate)
1983 prefix = type == static_cast<Char> ('b') ? "0b" : "0B";
1984 }
1985 else if (type == static_cast<Char> ('o'))
1986 {
1987 base = 8;
1988 if (specs.alternate && magnitude != 0)
1989 prefix = "0";
1990 }
1991 else if (type == static_cast<Char> ('x') || type == static_cast<Char> ('X'))
1992 {
1993 base = 16;
1994 if (specs.alternate)
1995 prefix = type == static_cast<Char> ('x') ? "0x" : "0X";
1996 }
1997 if (!specs.localized)
1998 {
1999 // Hot path without heap allocations: sign, prefix and digits are
2000 // built right-to-left in a stack array (128-bit binary fits).
2001 char const *const table = type == static_cast<Char> ('X')
2002 ? "0123456789ABCDEF"
2003 : "0123456789abcdef";
2004 Char text[140];
2005 Char *const end = text + sizeof (text) / sizeof (text[0]);
2006 Char *it = end;
2007 UInt value = magnitude;
2008 do
2009 {
2010 *--it = static_cast<Char> (
2011 table[static_cast<unsigned> (value % base)]);
2012 value = static_cast<UInt> (value / base);
2013 }
2014 while (value != 0);
2015 Char *const digits_begin = it;
2016 for (std::size_t i = prefix.size (); i-- > 0;)
2017 *--it = static_cast<Char> (prefix[i]);
2018 if (negative)
2019 *--it = static_cast<Char> ('-');
2020 else if (specs.sign == Sign::plus)
2021 *--it = static_cast<Char> ('+');
2022 else if (specs.sign == Sign::space)
2023 *--it = static_cast<Char> (' ');
2024 if (specs.zero_pad && specs.align == Align::none)
2025 {
2026 // ASCII only, so the width is the number of characters.
2027 std::size_t const length = static_cast<std::size_t> (end - it);
2028 std::size_t const target = static_cast<std::size_t> (specs.width);
2029 buffer.append (it, digits_begin);
2030 for (std::size_t i = length; i < target; ++i)
2031 buffer.push_back (static_cast<Char> ('0'));
2032 buffer.append (digits_begin, end);
2033 return;
2034 }
2035 write_padded (buffer, specs, Align::right, it, end);
2036 return;
2037 }
2038 std::basic_string<Char> digits = widen<Char> (
2039 to_base (magnitude, base, type == static_cast<Char> ('X')));
2040 // `L` groups the digits of every integer presentation (b, o, x too), as
2041 // std::format does; the base prefix stays in front of the groups.
2042 {
2043 std::numpunct<Char> const &facet
2044 = std::use_facet<std::numpunct<Char>> (ctx.locale ());
2045 digits = group_digits (digits, facet.grouping (), facet.thousands_sep ());
2046 }
2047 write_number (buffer, specs,
2048 sign_text<Char> (negative, specs.sign) + widen<Char> (prefix),
2049 digits, true);
2050}

2053template <typename Char>
2054std::basic_string<Char>
2055escape (Char const *it, Char const *end, Char quote)
2056{
2057 std::basic_string<Char> result (1, quote);
2058 while (it != end)
2059 {
2060 Char const *const start = it;
2061 bool valid = true;
2062 std::uint32_t const cp = decode (it, end, valid);
2063 if (!valid)
2064 {
2065 for (Char const *unit = start; unit != it; ++unit)
2066 {
2067 std::uint32_t const raw
2068 = sizeof (Char) == 1
2069 ? static_cast<unsigned char> (*unit)
2070 : static_cast<std::uint32_t> (*unit)
2071 & (sizeof (Char) == 2 ? 0xFFFFu : 0xFFFFFFFFu);
2072 result += widen<Char> ("\\x{" + to_base (raw, 16, false) + "}");
2073 }
2074 continue;
2075 }
2076 if (cp == '\t')
2077 result += widen<Char> ("\\t");
2078 else if (cp == '\n')
2079 result += widen<Char> ("\\n");
2080 else if (cp == '\r')
2081 result += widen<Char> ("\\r");
2082 else if (cp == '\\')
2083 result += widen<Char> ("\\\\");
2084 else if (cp == static_cast<std::uint32_t> (quote))
2085 {
2086 result.push_back (static_cast<Char> ('\\'));
2087 result.push_back (quote);
2088 }
2089 else if (cp < 0x20 || cp == 0x7F || (cp >= 0x80 && cp < 0xA0))
2090 result += widen<Char> ("\\u{" + to_base (cp, 16, false) + "}");
2091 else
2092 result.append (start, it);
2093 }
2094 result.push_back (quote);
2095 return result;
2096}

2381template <typename F>
2382std::string
2383float_body (F value, FloatStyle style, int precision, bool alternate,
2384 bool keep_zeros)
2385{
2386 std::string text;
2387 if (style == FloatStyle::shortest)
2388 text = shortest_decimal (value);
2389 else if (style == FloatStyle::general && keep_zeros)
2390 {
2391 text = printf_float (printf_pattern<F> ('g', true),
2392 precision == 0 ? 1 : precision, value);
2393 normalize_decimal_point (text);
2394 }
2395 else
2396 {
2397#if LUMEX_FORMAT_HAS_FLOAT_TO_CHARS
2398 std::chars_format const format
2399 = style == FloatStyle::general ? std::chars_format::general
2400 : style == FloatStyle::scientific ? std::chars_format::scientific
2401 : style == FloatStyle::fixed ? std::chars_format::fixed
2402 : std::chars_format::hex;
2403 int const digits
2404 = style == FloatStyle::general && precision == 0 ? 1 : precision;
2405 // A stack buffer covers every common case; only a large precision
2406 // (or a huge fixed value) needs the heap.
2407 char stack[512];
2408 std::to_chars_result const first_try
2409 = precision < 0
2410 ? std::to_chars (stack, stack + sizeof (stack), value, format)
2411 : std::to_chars (stack, stack + sizeof (stack), value, format,
2412 digits);

2414 text.assign (stack, first_try.ptr);
2415 else
2416 {
2417 std::vector<char> buffer (static_cast<std::size_t> (
2418 1024 + (precision > 0 ? precision : 0)));
2419 for (;;)
2420 {
2421 char *const first = buffer.data ();
2422 char *const stop = first + buffer.size ();
2423 std::to_chars_result const result
2424 = precision < 0
2425 ? std::to_chars (first, stop, value, format)
2426 : std::to_chars (first, stop, value, format, digits);
2427 if (result.ec == std::errc ())
2428 {
2429 text.assign (first, result.ptr);
2430 break;
2431 }
2432 buffer.resize (buffer.size () * 2);
2433 }
2434 }
2435#else
2436 if (style == FloatStyle::hex)
2437 {
2438 if (precision < 0)
2439 {
2440 char const *const pattern
2441 = std::is_same<F, long double>::value ? "%La" : "%a";
2442 typedef typename std::conditional<
2443 std::is_same<F, long double>::value, long double,
2444 double>::type promoted_type;
2445 char raw[128];
2446 std::snprintf (raw, sizeof (raw), pattern,
2447 static_cast<promoted_type> (value));
2448 text = raw;
2449 normalize_decimal_point (text);
2450 text = strip_hex_prefix (text, true);
2451 }
2452 else
2453 {
2454 text = printf_float (printf_pattern<F> ('a', false), precision,
2455 value);
2456 normalize_decimal_point (text);
2457 text = strip_hex_prefix (text, false);
2458 }
2459 }
2460 else
2461 {
2462 char const type = style == FloatStyle::general ? 'g'
2463 : style == FloatStyle::scientific ? 'e'
2464 : 'f';

2466 printf_pattern<F> (type, false),
2467 style == FloatStyle::general && precision == 0 ? 1 : precision,
2468 value);
2469 normalize_decimal_point (text);
2470 }
2471#endif
2472 }
2473 if (alternate && text.find ('.') == std::string::npos)
2474 {
2475 std::size_t const exponent
2476 = text.find_first_of (style == FloatStyle::hex ? "p" : "e");
2477 if (exponent == std::string::npos)
2478 text.push_back ('.');
2479 else
2480 text.insert (exponent, 1, '.');
2481 }
2482 return text;

2485template <typename Char, typename F>
2486void
2487write_float (Buffer<Char> &buffer, format_specs_t<Char> const &specs, F value,
2488 BasicFormatContext<Char> const &ctx)
2489{
2490 Char const type = specs.type;
2491 bool const upper
2492 = type == static_cast<Char> ('E') || type == static_cast<Char> ('F')
2493 || type == static_cast<Char> ('G') || type == static_cast<Char> ('A');
2494 bool const negative = std::signbit (value);
2495 std::basic_string<Char> const sign = sign_text<Char> (negative, specs.sign);
2496 if (!std::isfinite (value))
2497 {
2498 std::string text = std::isnan (value) ? "nan" : "inf";
2499 if (upper)
2500 for (std::size_t i = 0; i < text.size (); ++i)
2501 text[i] = static_cast<char> (text[i] - 'a' + 'A');
2502 write_number (buffer, specs, sign, widen<Char> (text), false);
2503 return;
2504 }
2505 F const magnitude = negative ? -value : value;
2506 FloatStyle style = FloatStyle::shortest;
2507 int precision = specs.precision;
2508 if (type == Char ())
2509 style = precision < 0 ? FloatStyle::shortest : FloatStyle::general;
2510 else if (type == static_cast<Char> ('a') || type == static_cast<Char> ('A'))
2511 style = FloatStyle::hex;
2512 else if (type == static_cast<Char> ('e') || type == static_cast<Char> ('E'))
2513 {
2514 style = FloatStyle::scientific;
2515 if (precision < 0)
2516 precision = 6;
2517 }
2518 else if (type == static_cast<Char> ('f') || type == static_cast<Char> ('F'))
2519 {
2520 style = FloatStyle::fixed;
2521 if (precision < 0)
2522 precision = 6;
2523 }
2524 else
2525 {
2526 style = FloatStyle::general;
2527 if (precision < 0)
2528 precision = 6;
2529 }
2530 // Fixed / scientific output grows with the precision: refuse a size that
2531 // cannot be represented rather than trying to allocate it.
2532 if ((style == FloatStyle::fixed || style == FloatStyle::scientific)
2533 && precision > INT_MAX - 1024)
2534 throw FormatError ("number is too big");
2535 // Past the number of significant digits a value can have, a larger
2536 // general precision prints the same text; cap it to bound the buffer.
2537 int const general_cap = std::numeric_limits<F>::digits
2538 - std::numeric_limits<F>::min_exponent + 16;
2539 if (style == FloatStyle::general && precision > general_cap)
2540 precision = general_cap;
2541 std::string text = float_body (magnitude, style, precision, specs.alternate,
2542 specs.alternate && type != Char ());
2543 if (upper)
2544 for (std::size_t i = 0; i < text.size (); ++i)
2545 if (text[i] >= 'a' && text[i] <= 'z')
2546 text[i] = static_cast<char> (text[i] - 'a' + 'A');
2547
2548 std::basic_string<Char> body = widen<Char> (text);
2549 if (specs.localized && style != FloatStyle::hex)
2550 {
2551 std::numpunct<Char> const &facet
2552 = std::use_facet<std::numpunct<Char>> (ctx.locale ());
2553 std::size_t integral = 0;
2554 while (integral < body.size () && is_digit (body[integral]))
2555 ++integral;
2556 std::basic_string<Char> rest = body.substr (integral);
2557 if (!rest.empty () && rest[0] == static_cast<Char> ('.'))
2558 rest[0] = facet.decimal_point ();
2559 body = group_digits (body.substr (0, integral), facet.grouping (),
2560 facet.thousands_sep ())
2561 + rest;
2562 }
2563 write_number (buffer, specs, sign, body, true);
2564}
