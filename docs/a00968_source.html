1581  // Find node from subtree using predicate. Returns first node from subtree
1582  // (depth-first), for which predicate returned true.
1598  template <typename Predicate>
1599  XmlNode
1600  find_node (Predicate pred) const
1601  {
1602  if (!m_root)
1603  return {};
1604 
1605  XmlNode cur = first_child ();
1606 
1607  while (cur.m_root && cur.m_root != m_root)
1608  {
1609  if (pred (cur))
1610  return cur;
1611 
1612  if (cur.first_child ())
1613  cur = cur.first_child ();
1614  else if (cur.next_sibling ())
1615  cur = cur.next_sibling ();
1616  else
1617  {
1618  while (!cur.next_sibling () && cur.m_root != m_root)
1619  cur = cur.parent ();
1620 
1621  if (cur.m_root != m_root)
1622  cur = cur.next_sibling ();
1623  }
1624  }
1625 
1626  return {};
1627  }
