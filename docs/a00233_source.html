225template <typename Char, typename WriteBody>
226void
227write_with_width (format_specs_t<Char> const &parsed,
228 BasicFormatContext<Char> &ctx, WriteBody const &write_body)
229{
230 format_specs_t<Char> specs = parsed;
231 specs.width = resolve_dynamic (specs.width_ref, specs.width, ctx);
232 if (specs.width == 0)
233 {
234 write_body (ctx);
235 return;
236 }
237 std::basic_string<Char> text;
238 StringBuffer<Char> buffer (text);
239 std::locale const locale = ctx.locale ();
240 BasicFormatContext<Char> inner (buffer, ctx.args (), &locale);
241 write_body (inner);
242 write_padded (ctx.out ().buffer (), specs, Align::left, text);
243}

444 Char const *
445 parse (BasicFormatParseContext<Char> &ctx)
446 {
447 _specs = Detail::format_specs_t<Char> (); // user brackets stay
448 Char const *it = Detail::parse_range_fill_align_width (ctx, _specs);
449 Char const *const end = ctx.end ();
450 if (it != end && *it == static_cast<Char> ('n'))
451 {
452 _open.clear ();
453 _close.clear ();
454 ++it;
455 }
456 if (it != end && *it == static_cast<Char> ('m'))
457 {
458 if (!lumex::core::utility::traits::tuple::is_pair_like<
459 element_type>::value)
460 throw FormatError ("invalid format specifier");
461 if (!_open.empty ())
462 {
463 _open = Detail::widen<Char> ("{");
464 _close = Detail::widen<Char> ("}");
465 }
466 _presentation = presentation::map;
467 ++it;
468 }
469 else if (it != end && *it == static_cast<Char> ('s'))
470 {
471 if (!std::is_same<element_type, Char>::value)
472 throw FormatError ("invalid format specifier");
473 _presentation = presentation::text;
474 ++it;
475 }
476 else if (it != end && *it == static_cast<Char> ('?') && end - it > 1
477 && it[1] == static_cast<Char> ('s'))
478 {
479 if (!std::is_same<element_type, Char>::value)
480 throw FormatError ("invalid format specifier");
481 _presentation = presentation::debug_text;
482 it += 2;
483 }
484 if (it != end && *it == static_cast<Char> (':'))
485 {
486 if (_presentation == presentation::text
487 || _presentation == presentation::debug_text)
488 throw FormatError ("invalid format specifier");
489 ctx.advance_to (it + 1);
490 it = _underlying.parse (ctx);
491 }
492 else
493 Detail::parse_default_element<Char, element_type> (_underlying);
494 if (it != end && *it != static_cast<Char> ('}'))
495 throw FormatError ("invalid format specifier");
496 if (_presentation == presentation::map)
497 Detail::use_map_element_style (_underlying, 0);
498 return it;
499 }

566 void
567 write_body (Range const &range, BasicFormatContext<Char> &ctx) const
568 {
569 Detail::Buffer<Char> &buffer = ctx.out ().buffer ();
570 if (_presentation == presentation::text
571 || _presentation == presentation::debug_text)
572 {
573 write_text (range, buffer, std::is_same<element_type, Char> ());
574 return;
575 }
576 buffer.append (_open);
577 bool first = true;
578 for (auto it = std::begin (range); it != std::end (range); ++it)
579 {
580 if (!first)
581 buffer.append (_separator);
582 first = false;
583 _underlying.format (*it, ctx);
584 }
585 buffer.append (_close);
586 }
