843 template <typename Allocator>
844 std::string
845 to_string (LumexBasicStacktrace<Allocator> const &stacktrace)
846 {
847  std::string result;
848  result.reserve (stacktrace.size ()
849  * detail::kDefaultMaxFrames); // Pre-allocate for efficiency
850 
851  for (std::size_t i = 0; i < stacktrace.size (); ++i)
852  {
853  result += std::to_string (i);
854  result += "# ";
855  result += stacktrace[i].description ();
856  result += '\n';
857  }
858 
859  return result;
860 }

894 template <typename Allocator> struct hash<LumexBasicStacktrace<Allocator>>
895 {
902  std::size_t
903  operator() (LumexBasicStacktrace<Allocator> const &stacktrace) const
904  LUMEX_NOEXCEPT
905  {
906  std::size_t seed = 0;
907  for (auto const &entry : stacktrace)
908  {
909  seed ^= std::hash<void *> () (entry.native_handle ())
910  + detail::kHashGoldenRatio + (seed << detail::kHashLeftShift)
911  + (seed >> detail::kHashRightShift);
912  }
913  return seed;
914  }
