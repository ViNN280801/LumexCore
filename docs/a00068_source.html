1100 reference
1101 emplace_back(Args &&...args)
1102 {
1103 if(full())
1104 {
1105 // эквивалент pop_front(): удалить старейший, сделать буфер неполным
1106 alloc_traits::destroy(m_allocator, std::addressof(at_slot(0)));
1107 m_head = next_index(m_head);
1108 --m_size;
1109 }

1111 alloc_traits::construct(m_allocator, reinterpret_cast<T *>(slot_ptr(pos)), std::forward<Args>(args)...);

1131 reference
1132 emplace_front(Args &&...args)
1133 {
1134 size_type const pos = (m_size == 0) ? m_head : prev_index(m_head);
1135
1136 if(full())
1137 {
1138 // освободить целевой слот: это именно prev_index(m_head)
1139 alloc_traits::destroy(m_allocator, std::addressof(at_slot(0 + m_size - 1))); // старый самый новый
