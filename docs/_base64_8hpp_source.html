93 // This function specifically excludes '=' as a valid alphabet character, as '=' is padding.

200 template <typename T>
201 std::string
202 _encode_impl(T const &data_view)
203 {
204 static_assert(
205 has_convertible_size<T>::value,
206 "Encoding data requires a type with a .size() method whose return type is convertible to size_t.");
207 static_assert(has_convertible_indexed_access<T>::value,
208 "Encoding data requires a type with an operator[] that accepts size_t as an index and returns "
209 "a type convertible to byte_type.");
210
211 if(data_view.empty()) return {};
212
213 // Calculate output length and reserve space
214 std::string result;
215 size_t out_length{4 * ((data_view.size() + 2) / 3)};
216 result.reserve(out_length);
217
218 // 1. Process 3-byte chunks
219 size_t pos{};
220 while(pos + 2 < data_view.size())
221 {
222 // 1.1. Get the first 6 bits of the first byte
223 result += detail::_base64_chars.at((data_view[pos] >> 2) & Constants::kBase64MaskSixBits);
224
225 // 1.2. Get the last 2 bits of the first byte and the first 4 bits of the second byte
226 result += detail::_base64_chars.at(((data_view[pos] & Constants::kBase64MaskTwoBits) << 4)
227 | ((data_view[pos + 1] >> 4) & Constants::kBase64MaskFourBits));
228
229 // 1.3. Get the last 4 bits of the second byte and the first 2 bits of the third byte
230 result += detail::_base64_chars.at(
231 ((data_view[pos + 1] & Constants::kBase64MaskFourBits) << 2)
232 | ((data_view[pos + 2] >> Constants::kBase64RightShiftSixBits) & Constants::kBase64MaskTwoBits));
233
234 // 1.4. Get the last 6 bits of the third byte
235 result += detail::_base64_chars.at(data_view[pos + 2] & Constants::kBase64MaskSixBits);
236
237 // 1.5. Move to the next 3 bytes to process
238 pos += 3;
239 }
240
241 // 2. Handle remaining bytes
242 if(pos < data_view.size())
243 {
244 // 2.1. Get the first 6 bits of the first byte
245 result += detail::_base64_chars.at((data_view[pos] >> 2) & Constants::kBase64MaskSixBits);
246
247 if(pos + 1 < data_view.size())
248 {
249 // 2.1.1. Get the last 2 bits of the first byte and the first 4 bits of the second byte
250 result += detail::_base64_chars.at(((data_view[pos] & Constants::kBase64MaskTwoBits) << 4)
251 | ((data_view[pos + 1] >> 4) & Constants::kBase64MaskFourBits));
252
253 // 2.1.2. Get the last 4 bits of the second byte
254 result += detail::_base64_chars.at((data_view[pos + 1] & Constants::kBase64MaskFourBits) << 2);
255
256 // 2.1.3. Add padding
257 result += '=';
258 }
259 else
260 {
261 // 2.2.1. Get the last 2 bits of the first byte and the first 4 bits of the second byte
262 result += detail::_base64_chars.at((data_view[pos] & Constants::kBase64MaskTwoBits) << 4);
263
264 // 2.2.2. Add double padding, because there is only one byte left
265 result += "==";
266 }
267 }
268
269 // 3. Return the result
270 return result;
271 }
