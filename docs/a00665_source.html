117 // Packing control macros for MSVC compatibility
118 // @details These macros provide portable packing control across GCC/Clang and
119 // MSVC
120 // @note GCC/Clang support #pragma pack for MSVC compatibility, so these macros
121 // work on all platforms
122 // @usage:
123 // LUMEX_PACK_BEGIN(1)
124 // struct MyStruct {
125 // char c;
126 // int i;
127 // };
128 // LUMEX_PACK_END
129 #if defined(_MSC_VER)
130 #define LUMEX_PACK_BEGIN(alignment) __pragma (pack (push, alignment))
131 #define LUMEX_PACK_END __pragma (pack (pop))
132 #elif defined(__GNUC__) || defined(__clang__)
133 // GCC/Clang support MSVC-style #pragma pack for compatibility
134 // GCC 4.9+ and Clang support __pragma (same as MSVC)
135 #if (defined(__GNUC__) \
136  && (__GNUC__ >= 5 || (__GNUC__ == 4 && __GNUC_MINOR__ >= 9))) \
137  || defined(__clang__)
138 #define LUMEX_PACK_BEGIN(alignment) __pragma (pack (push, alignment))
139 #define LUMEX_PACK_END __pragma (pack (pop))
140 #else
141 // Fallback to _Pragma for older GCC versions
142 // Define common alignment values directly
143 #define LUMEX_PACK_BEGIN_1 _Pragma ("pack(push, 1)")
144 #define LUMEX_PACK_BEGIN_2 _Pragma ("pack(push, 2)")
145 #define LUMEX_PACK_BEGIN_4 _Pragma ("pack(push, 4)")
146 #define LUMEX_PACK_BEGIN_8 _Pragma ("pack(push, 8)")
147 #define LUMEX_PACK_BEGIN(alignment) LUMEX_PACK_BEGIN_##alignment
148 #define LUMEX_PACK_END _Pragma ("pack(pop)")
149 #endif
150 #else
151 #define LUMEX_PACK_BEGIN(alignment)
152 #define LUMEX_PACK_END
153 #endif

202 // [[likely]] / [[unlikely]]
203 // LUMEX_ATTRIBUTE_LIKELY_COND(cond) and LUMEX_ATTRIBUTE_UNLIKELY_COND(cond)
204 // wrap a condition and exist in every standard. Below C++20 they pass the hint
205 // through __builtin_expect on GCC and Clang (the condition is converted to
206 // bool first, so a pointer works too) and are the plain condition elsewhere.
207 // The statement attributes LUMEX_ATTRIBUTE_LIKELY and LUMEX_ATTRIBUTE_UNLIKELY
208 // are empty below C++20: no compiler has a GNU spelling for them (GCC rejects
209 // __attribute__ ((likely)) before a statement, Clang ignores it with a
210 // warning).
211 #if __cplusplus >= 202002L
212 #define LUMEX_ATTRIBUTE_LIKELY [[likely]]
213 #define LUMEX_ATTRIBUTE_UNLIKELY [[unlikely]]
214 #define LUMEX_ATTRIBUTE_LIKELY_COND(cond) (cond)
215 #define LUMEX_ATTRIBUTE_UNLIKELY_COND(cond) (cond)
216 #else
217 #define LUMEX_ATTRIBUTE_LIKELY
218 #define LUMEX_ATTRIBUTE_UNLIKELY
219 #if defined(__GNUC__) || defined(__clang__)
220 #define LUMEX_ATTRIBUTE_LIKELY_COND(cond) (__builtin_expect (!!(cond), 1))
221 #define LUMEX_ATTRIBUTE_UNLIKELY_COND(cond) (__builtin_expect (!!(cond), 0))
222 #else
223 #define LUMEX_ATTRIBUTE_LIKELY_COND(cond) (cond)
224 #define LUMEX_ATTRIBUTE_UNLIKELY_COND(cond) (cond)
225 #endif
226 #endif
