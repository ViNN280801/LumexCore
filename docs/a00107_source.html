744 size_t
745 operator()(LumexBasicStacktrace<Allocator> const &stacktrace) const noexcept
746 {
747 size_t seed = 0;
748 for(auto const &entry : stacktrace)
749 {
750 seed ^= std::hash<void *>()(entry.native_handle()) + detail::kHashGoldenRatio
751 + (seed << detail::kHashLeftShift) + (seed >> detail::kHashRightShift);
752 }
