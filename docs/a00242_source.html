377template <typename It1, typename End1, typename It2, typename End2>
378void
379require_same_size (char const *function, It1 &it1, End1 const &last1, It2 &it2,
380 End2 const &last2, std::size_t count)
381{
382 if (!(it1 != last1) && !(it2 != last2))
383 return;
384
385 std::size_t size1 = count;
386 std::size_t size2 = count;
387 for (; it1 != last1; ++it1)
388 ++size1;
389 for (; it2 != last2; ++it2)
390 ++size2;
391 throw LumexMathSizeMismatchException (
392 std::string (function) + ": size of both ranges must be equal ("
393 + std::to_string (size1) + " vs " + std::to_string (size2) + ")");
394}
