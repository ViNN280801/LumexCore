238 T
239 generate_number (T from_val, T to_val, DistributionType dist_type) const
240 {
241 switch (dist_type)
242 {
243 case DistributionType::UNIFORM:
244 return generate_uniform (from_val, to_val);
245
246 case DistributionType::NORMAL:
247 return generate_normal_safe (from_val, to_val);
248
249 case DistributionType::EXPONENTIAL:
250 return generate_exponential_safe (from_val);
251
252 case DistributionType::GAMMA:
253 return generate_gamma_safe (from_val, to_val);
254
255 case DistributionType::BERNOULLI:
256 return generate_bernoulli_safe (static_cast<double> (from_val));
257
258 case DistributionType::BINOMIAL:
259 return generate_binomial_safe (static_cast<int> (to_val),
260 static_cast<double> (from_val));
261
262 case DistributionType::GEOMETRIC:
263 return generate_geometric_safe (static_cast<double> (from_val));
264
265 case DistributionType::POISSON:
266 return generate_poisson_safe (static_cast<double> (from_val));
267
268 default:
269 return generate_uniform (from_val, to_val);
270 }
271 }

984 std::vector<T>
985 get_sequence (std::size_t count, T from_val = get_default_min (),
986 T to_val = get_default_max (),
987 DistributionType dist_type = DistributionType::UNIFORM) const
988 {
989 if (count == 0)
990 {
991 // Note: Returning empty sequence for zero count request
992 return {};
993 }
994
995 try
996 {
997 std::vector<T> sequence;
998 sequence.reserve (count); // Pre-allocate memory to avoid reallocations
999
1000 for (std::size_t i = 0; i < count; ++i)
1001 sequence.push_back (generate_number (from_val, to_val, dist_type));
1002
1003 return sequence;
1004 }
1005 catch (std::exception const &e)
1006 {
1007 std::cerr << "Error generating sequence: " << e.what ()
1008 << ". Parameters: count=" << count << ", from=" << from_val
1009 << ", to=" << to_val << ". Returning empty sequence.\n";
1010 return {};
1011 }
1012 catch (...)
1013 {
1014 std::cerr << "Unknown exception during sequence generation. "
1015 << "Parameters: count=" << count << ", from=" << from_val
1016 << ", to=" << to_val << ". Returning empty sequence.\n";
1017 return {};
1018 }
1019 }
