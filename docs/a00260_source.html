221 template <typename T>
222 std::string
223 VarInfo (char const *exprText, T &&value, bool isExprNoexcept,
224  char const *file, int line)
225 {
226  // The address reuses exactly the pointer type addressof returned
227  // (via decltype), not one rebuilt from decay<T> - for an
228  // array/string literal, addressof yields a pointer TO THE ARRAY
229  // (e.g. int(*)[5]), while decay<T>* would give int** - a completely
230  // different, incompatible pointer structure (const_cast between them
231  // is ill-formed, not merely imprecise). remove_pointer+remove_cv
232  // strips the qualification while keeping the original (possibly
233  // array) structure, and that structure - not the decay-simplified
234  // one - is the one valid for a static_cast to an object pointer ->
235  // void* ([conv.ptr] requires a pointer-to-object-type, which
236  // pointer-to-array is, unlike the incompatible "pointer to element").
237  // typename ...::type, not remove_cv_t/remove_pointer_t: this function
238  // (unlike VarInfoDetail::FormatValue above) is the common C++11
239  // baseline, and the _t aliases only appeared in C++14.
240  using AddressPointerType = decltype (std::addressof (value));
241  using RawPointerType = typename std::remove_cv<
242  typename std::remove_pointer<AddressPointerType>::type>::type *;
243  void const *addr = static_cast<void const *> (
244  const_cast<RawPointerType> (std::addressof (value)));
245 
246  // sizeof(value), not sizeof(decay<T>::type): for an array, decay
247  // would collapse the type into a pointer and sizeof would return the
248  // pointer's size (8 bytes), not the array's real size - sizeof on a
249  // reference parameter transparently gives the size of what it refers to.
250  std::ostringstream oss;
251  oss << exprText << " = " << VarInfoDetail::FormatValue (value) << " [type="
252  << ::lumex::core::utility::demangle::demangle_type_name (
253  typeid (value).name ())
254  << (std::is_const<typename std::remove_reference<T>::type>::value
255  ? " const"
256  : "")
257  << (std::is_volatile<typename std::remove_reference<T>::type>::value
258  ? " volatile"
259  : "")
260  << ", size=" << sizeof (value) << " bytes, addr=" << addr
261  << ", noexcept=" << (isExprNoexcept ? "true" : "false") << ", at "
262  << file << ":" << line << "]";
263  return oss.str ();
264 }
