33 // ====================== Static assertions ====================== //
34 // Запрещаем быть ссылочным типом, так как:
35 // 1. Данный класс (и его специализации) предназначены для
36 // владения хранимым значением - успешным или ошибочным. Если бы типы SuccessType или ErrorType были
37 // ссылочными, например `int&`, то Expected не владел бы объектом, а лишь ссылался бы на него.
38 // Такое поведение может привести к висячим ссылкам, если оригинальный объект, на который ссылается
39 // SuccessType (или ErrorType), будет уничтожен раньше, чем Expected.
40 // 2. Внутренние ограничения типа `union`: стандарт С++ запрещает членам `union` быть ссылочными типами,
41 // так как ссылки не могут быть перемещаемыми или копируемыми.
42 // 3. Соответствие C++23: стандартная библиотека std::expected из C++23 также явно запрещает ссылочные типы для
43 // параметра ошибки по тем же причинам. Это обеспечивает согласованность и предсказуемость поведения.
44 static_assert(!std::is_reference<SuccessType>::value, "Expected<T,E>: T must not be a reference");
45 static_assert(!std::is_function<SuccessType>::value, "Expected<T,E>: T must not be a function type");

390 LUMEX_ATTRIBUTE_NODISCARD("Return value is the contained error; should always be used.")
391 LUMEX_CONSTEXPR_FUNCTION ErrorType &
392 error() &
393 {
394 // Предусловие: !m_has_value.
395 // Нарушение - UB (неопределенное поведение) по двум причинам:
396 //
397 // 1) Контракт уровня интерфейса (совместимость со стандартом C++23).
398 // В модели std::expected метод error() - непроверенный наблюдатель с предусловием,
399 // что объект находится в состоянии ошибки. При нарушении предусловия поведение не
400 // определено. Проверенный доступ предусмотрен через value(), который бросает
401 // bad_expected_access<E>. Мы сознательно сохраняем ту же модель: error() не делает
402 // дополнительных проверок/бросков, чтобы оставаться noexcept и без накладных расходов.
403 //
404 // 2) Правила языка для union (уровень реализации).
405 // Внутри Expected хранение реализовано через union: либо активен член со значением
406 // (или фиктивный маркер для void-успеха), либо активен член m_error. Если m_has_value == true,
407 // активен НЕ m_error. Любой доступ к неактивному члену объединения - это UB по стандарту C++.
408 // Следовательно, чтение m_storage.m_error при m_has_value == true уже само по себе
409 // приводит к неопределенному поведению на уровне языка.
410 //
411 // Почему не бросаем исключение здесь?
412 // - Метод должен оставаться пригодным для noexcept-контекстов (деструкторы, swap, аварийные пути).
413 // - Нулевые накладные расходы в релизе: assert удаляется под NDEBUG, не добавляя ветвлений/бросков.
414 // - Проверяемый, перехватываемый доступ уже предоставлен через value() (и предварительную
415 // проверку has_value()).
416 //
417 // Практическое следствие:
418 // - В отладке assert упадет сразу, показывая место нарушения контракта.
419 // - В релизе ответственность на вызывающей стороне: перед вызовом error() необходимо
420 // гарантировать !has_value() (например, через if(!has_value()) ...).
421 assert(!m_has_value && "Calling error() while value is present is undefined behavior.");
422 return m_storage.m_error;
423 }
