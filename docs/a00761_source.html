410template <typename T>
411LUMEX_API inline T *
412new_xpath_variable (char_t const *name)
413{
414 std::size_t length = utility::strlength (name);
415 if (length == 0)
416 return nullptr; // empty variable names are invalid
417
418 // $$ we can't use offsetof(T, name) because T is non-POD, so we just
419 // allocate additional length characters
420 void *memory // NOLINT(cppcoreguidelines-owning-memory)
421 = malloc (sizeof (T) // NOLINT(cppcoreguidelines-no-malloc)
422 + (length * sizeof (char_t)));
423 if (!memory)
424 return nullptr;
425
426 T *result = new (memory) T (); // NOLINT(cppcoreguidelines-owning-memory)
427
428 std::memcpy (
429 result
430 ->name, // NOLINT(cppcoreguidelines-pro-bounds-array-to-pointer-decay)
431 name, (length + 1) * sizeof (char_t));
432
433 return result;
434}
