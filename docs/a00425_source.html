306 new_xpath_variable(char_t const *name)
307 {
308 size_t length = Utility::strlength(name);
309 if(length == 0) return nullptr; // empty variable names are invalid
310
311 // $$ we can't use offsetof(T, name) because T is non-POD, so we just allocate additional length characters
312 void *memory // NOLINT(cppcoreguidelines-owning-memory)
313 = malloc(sizeof(T) // NOLINT(cppcoreguidelines-no-malloc)
314 + (length * sizeof(char_t)));
315 if(!memory) return nullptr;
316
317 T *result = new(memory) T(); // NOLINT(cppcoreguidelines-owning-memory)
318
319 std::memcpy(result->name, // NOLINT(cppcoreguidelines-pro-bounds-array-to-pointer-decay)
320 name, (length + 1) * sizeof(char_t));
321
322 return result;
323 }
