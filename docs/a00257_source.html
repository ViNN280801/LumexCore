219template <typename T>
220std::string
221VarInfo (char const *exprText, T &&value, bool isExprNoexcept,
222 char const *file, int line)
223{
224 // The address reuses exactly the pointer type addressof returned
225 // (via decltype), not one rebuilt from decay<T> - for an
226 // array/string literal, addressof yields a pointer TO THE ARRAY
227 // (e.g. int(*)[5]), while decay<T>* would give int** - a completely
228 // different, incompatible pointer structure (const_cast between them
229 // is ill-formed, not merely imprecise). remove_pointer+remove_cv
230 // strips the qualification while keeping the original (possibly
231 // array) structure, and that structure - not the decay-simplified
232 // one - is the one valid for a static_cast to an object pointer ->
233 // void* ([conv.ptr] requires a pointer-to-object-type, which
234 // pointer-to-array is, unlike the incompatible "pointer to element").
235 // typename ...::type, not remove_cv_t/remove_pointer_t: this function
236 // (unlike VarInfoDetail::FormatValue above) is the common C++11
237 // baseline, and the _t aliases only appeared in C++14.
238 using AddressPointerType = decltype (std::addressof (value));
239 using RawPointerType = typename std::remove_cv<
240 typename std::remove_pointer<AddressPointerType>::type>::type *;
241 void const *addr = static_cast<void const *> (
242 const_cast<RawPointerType> (std::addressof (value)));
243
244 // sizeof(value), not sizeof(decay<T>::type): for an array, decay
245 // would collapse the type into a pointer and sizeof would return the
246 // pointer's size (8 bytes), not the array's real size - sizeof on a
247 // reference parameter transparently gives the size of what it refers to.
248 std::ostringstream oss;
249 oss << exprText << " = " << VarInfoDetail::FormatValue (value) << " [type="
250 << ::lumex::core::utility::demangle::demangle_type_name (
251 typeid (value).name ())
252 << (std::is_const<typename std::remove_reference<T>::type>::value
253 ? " const"
254 : "")
255 << (std::is_volatile<typename std::remove_reference<T>::type>::value
256 ? " volatile"
257 : "")
258 << ", size=" << sizeof (value) << " bytes, addr=" << addr
259 << ", noexcept=" << (isExprNoexcept ? "true" : "false") << ", at "
260 << file << ":" << line << "]";
261 return oss.str ();
262}
