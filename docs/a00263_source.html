1027 template <typename Predicate>

1039 // Find node from subtree using predicate. Returns first node from subtree (depth-first), for which predicate
1040 // returned true.
1054 template <typename Predicate>
1055 XmlNode
1056 find_node(Predicate pred) const
1057 {
1058 if(!m_root) return {};
1059
1060 XmlNode cur = first_child();
1061
1062 while(cur.m_root && cur.m_root != m_root)
1063 {
1064 if(pred(cur)) return cur;
1065
1066 if(cur.first_child())
1067 cur = cur.first_child();
1068 else if(cur.next_sibling())
1069 cur = cur.next_sibling();
1070 else
1071 {
1072 while(!cur.next_sibling() && cur.m_root != m_root) cur = cur.parent();
1073
1074 if(cur.m_root != m_root) cur = cur.next_sibling();
1075 }
1076 }
1077
1078 return {};
1079 }
