496 optional &
497 operator= (optional const &other)
498 LUMEX_NOEXCEPT_IF (std::is_nothrow_copy_constructible<T>::value
499 &&std::is_nothrow_copy_assignable<T>::value)
500 {
501 if (this != &other) // Handle self-assignment
502 {

504 {
505 if (m_has_value) // '*this' also has a value: assign or destroy
506 // old, construct new
507 {
508 // If T is trivially copy assignable, we could assign directly.
509 // For general case, destroy and reconstruct is safer for
510 // non-trivial types. This implementation chooses safety over
511 // potential micro-optimizations.
512 destroy ();
513 construct (*other.get_ptr ());
514 }

518 }
519 }
