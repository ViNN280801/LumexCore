367 LumexOptional &
368 operator=(LumexOptional const &other) noexcept(std::is_nothrow_copy_constructible<T>::value
369 && std::is_nothrow_copy_assignable<T>::value)
370 {
371 if(this != &other) // Handle self-assignment
372 {
373 if(other.m_has_value) // 'other' has a value
374 {
375 if(m_has_value) // '*this' also has a value: assign or destroy old, construct new
376 {
377 // If T is trivially copy assignable, we could assign directly.
378 // For general case, destroy and reconstruct is safer for non-trivial types.
379 // This implementation chooses safety over potential micro-optimizations.
380 destroy();
381 construct(*other.get_ptr());
382 }
383 else // '*this' is empty, 'other' has value: construct new
384 {
385 construct(*other.get_ptr());
386 }
387 }
388 else // 'other' is empty
389 {
390 if(m_has_value) // '*this' has a value, 'other' is empty: destroy '*this' value
391 destroy();
392 // else: Both are empty, nothing to do.
393 }
394 }
395 return *this;
396 }

664 swap(LumexOptional &other) noexcept(std::is_nothrow_move_constructible<T>::value
665 && std::is_nothrow_move_assignable<T>::value)
666 {
667 if(m_has_value && other.m_has_value)
668 {
669 using std::swap;
670 swap(**this, *other); // Swap contained values
671 }
672 else if(m_has_value)
673 {
674 // This has value, other is empty. Move this's value to other.
675 other.construct(std::move(**this));
676 destroy();
677 }
678 else if(other.m_has_value)
679 {
680 // Other has value, this is empty. Move other's value to this.
681 construct(std::move(*other));

683 }
