178 class BitLock
179 {
180 public:
184  enum : std::uint32_t
185  {
187  lock_bit = 1u,
190  notify_bit = 2u
191  };
192 
196  LUMEX_CONSTEXPR
197  BitLock () LUMEX_NOEXCEPT : word_ (0u) {}
198 
202  ~BitLock () { tsan_mutex_destroy (static_cast<void *> (&word_)); }
203 
204  BitLock (BitLock const &) = delete;
205  BitLock &operator= (BitLock const &) = delete;

212  void
213  lock () const LUMEX_NOEXCEPT
214  {
215  LUMEX_CONSTEXPR int relax_rounds = 12;
216  LUMEX_CONSTEXPR int yield_rounds = 4;
217  void *const address = static_cast<void *> (&word_);
218  tsan_mutex_pre_lock (address);
219  int round = 0;
220  std::uint32_t state = word_.load (std::memory_order_relaxed);
221  for (;;)
222  {
223  if ((state & lock_bit) == 0u)
224  {
225  if (word_.compare_exchange_weak (state, state | lock_bit,
226  std::memory_order_acquire,
227  std::memory_order_relaxed))
228  break;
229  continue;
230  }
231  if (round < relax_rounds + yield_rounds)
232  {
233  if (round < relax_rounds)
234  cpu_relax ();
235  else
236  std::this_thread::yield ();
237  ++round;
238  state = word_.load (std::memory_order_relaxed);
239  continue;
240  }
241  // Sleep only on a word that is taken and carries the sleeper bit:
242  // the holder then wakes the sleepers when it releases the lock. The
243  // bit is set again on every round, because each release clears it.
244  if ((state & notify_bit) == 0u)
245  {
246  if (!word_.compare_exchange_weak (state, state | notify_bit,
247  std::memory_order_relaxed,
248  std::memory_order_relaxed))
249  continue;
250  state |= notify_bit;
251  }
252  wait_until_changed (word_, state);
253  state = word_.load (std::memory_order_relaxed);
254  }
255  tsan_mutex_post_lock (address);
256  }
257 
264  void
265  unlock () const LUMEX_NOEXCEPT
266  {
267  void *const address = static_cast<void *> (&word_);
268  tsan_mutex_pre_unlock (address);
269  std::uint32_t const previous
270  = word_.exchange (0u, std::memory_order_release);
271  if ((previous & notify_bit) != 0u)
272  notify_all (word_);
273  tsan_mutex_post_unlock (address);
274  }

281  std::uint32_t
282  state () const LUMEX_NOEXCEPT
283  {
284  return word_.load (std::memory_order_relaxed);
285  }
286 
287 private:
288  mutable std::atomic<std::uint32_t> word_;
289 };
