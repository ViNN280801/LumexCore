310 // dl_iterate_phdr callback: the module whose loadable segment holds the
311 // address. dlpi_addr is the load bias, 0 for an executable linked at a fixed
312 // address, so address - bias is the address in the ELF file for executables
313 // (PIE or not) and shared libraries alike.
314 static inline int
315 find_module_callback (dl_phdr_info *info, std::size_t,
316  void *data) LUMEX_NOEXCEPT
317 {
318  try
319  {
320  auto *search = static_cast<module_search_t *> (data);
321  for (ElfW (Half) i = 0; i < info->dlpi_phnum; ++i)
322  {
323  ElfW (Phdr) const &segment = info->dlpi_phdr[i];
324  if (segment.p_type != PT_LOAD)
325  continue;
326  std::uintptr_t const start
327  = static_cast<std::uintptr_t> (info->dlpi_addr)
328  + static_cast<std::uintptr_t> (segment.p_vaddr);
329  if (search->address < start
330  || search->address
331  >= start + static_cast<std::uintptr_t> (segment.p_memsz))
332  continue;
333  search->result->found = true;
334  search->result->offset
335  = search->address
336  - static_cast<std::uintptr_t> (info->dlpi_addr);
337  char const *name = info->dlpi_name;
338  search->result->module = file_name_of (
339  name != nullptr && name[0] != '\0' ? std::string (name)
340  : executable_file_name ());
341  return 1;
342  }
343  return 0;
344  }
345  catch (...)
346  {
347  return 1;
348  }
349 }

352 // The module that holds address and the address inside it. Linux walks the
353 // loaded modules (dl_iterate_phdr); other systems fall back to dladdr and the
354 // module's base address, which is right for position-independent images.
355 static inline module_address_t
356 find_module_address (void const *address) LUMEX_NOEXCEPT
357 {

377  {
378  result = module_address_t ();

493 static inline std::string
494 captureStackTrace (int skip_frames = 1, int max_frames = 16) LUMEX_NOEXCEPT
495 {
496  LUMEX_CONSTEXPR int kMaxStackFrames = 64;
497  std::string result;
498 
499  try
500  {
501 #if defined(LUMEX_OS_WINDOWS)
502  // Windows: DbgHelp API for symbol information
503  void *stack[kMaxStackFrames];
504 
505  HANDLE process = GetCurrentProcess ();
506 
507  // Thread-safe symbol initialization (lazy, once)
508  if (!Detail::ensureSymbolsInitialized (process))
509  {
510  // Fallback: only addresses without symbols
511  WORD frames = CaptureStackBackTrace (
512  static_cast<DWORD> (skip_frames),
513  static_cast<DWORD> (max_frames), stack, nullptr);
514 
515  for (WORD i = 0; i < frames; ++i)
516  result += lumex::core::string::utility::stringify (
517  " #", i, ": [0x",
518  Detail::formatHex (reinterpret_cast<uintptr_t> (stack[i])),
519  "]\n");
520  return result.empty ()
521  ? " Stack trace unavailable (SymInitialize failed)\n"
522  : result;
523  }
524 
525  // Capture stack with full symbol resolution
526  WORD frames = CaptureStackBackTrace (static_cast<DWORD> (skip_frames),
527  static_cast<DWORD> (max_frames),
528  stack, nullptr);
529 
530  if (frames == 0)
531  return " Stack trace empty (no frames captured)\n";
532 
533  // Allocate symbol buffer (variable size structure)
534  LUMEX_CONSTEXPR std::size_t kSymbolBufferSize
535  = sizeof (SYMBOL_INFO) + (MAX_SYM_NAME * sizeof (TCHAR));
536  auto *symbol_buffer
537  = static_cast<SYMBOL_INFO *> (malloc (kSymbolBufferSize));
538 
539  if (symbol_buffer == nullptr)
540  return " Stack trace unavailable (memory allocation failed)\n";
541 
542  symbol_buffer->MaxNameLen = MAX_SYM_NAME;
543  symbol_buffer->SizeOfStruct = sizeof (SYMBOL_INFO);
544 
545  // Thread-safe symbol resolution
546  std::lock_guard<std::mutex> lock (Detail::getDbgHelpMutex ());
547 
548  // Refresh module list to include DLLs loaded after SymInitialize (e.g.
549  // plugins loaded at runtime). Ensures SymFromAddr can resolve addresses
550  // in dynamically loaded modules.
551  SymRefreshModuleList (process);
552 
553  for (WORD i = 0; i < frames; ++i)
554  {
555  DWORD64 address = reinterpret_cast<DWORD64> (stack[i]);
556 
557  // Get function name
558  DWORD64 displacement = 0;
559  if (SymFromAddr (process, address, &displacement, symbol_buffer))
560  {
561  // Try to get line number and file name
562  IMAGEHLP_LINE64 line{};
563  line.SizeOfStruct = sizeof (IMAGEHLP_LINE64);
564  DWORD line_displacement = 0;
565 
566  if (SymGetLineFromAddr64 (process, address, &line_displacement,
567  &line))
568  {
569  // Full info: function + file:line + address
570  result += lumex::core::string::utility::stringify (
571  " #", i, ": ", symbol_buffer->Name, " (", line.FileName,
572  ":", line.LineNumber, ") [0x",
573  Detail::formatHex (address), "]\n");
574  }
575  else
576  {
577  // Only function name + address (no line info)
578  result += lumex::core::string::utility::stringify (
579  " #", i, ": ", symbol_buffer->Name, " [0x",
580  Detail::formatHex (address), "]\n");
581  }
582  }
583  else
584  {
585  // Fallback: try SymGetModuleInfo64 for module name (e.g.
586  // ntdll.dll+0x1234)
587  IMAGEHLP_MODULE64 moduleInfo{};
588  moduleInfo.SizeOfStruct = sizeof (IMAGEHLP_MODULE64);
589  if (SymGetModuleInfo64 (process, address, &moduleInfo))
590  {
591  DWORD64 const offset = address - moduleInfo.BaseOfImage;
592  result += lumex::core::string::utility::stringify (
593  " #", i, ": ", moduleInfo.ModuleName, "+0x",
594  Detail::formatHex (offset), " [0x",
595  Detail::formatHex (address), "]\n");
596  }
597  else
598  {
599  result += lumex::core::string::utility::stringify (
600  " #", i, ": [0x", Detail::formatHex (address), "]\n");
601  }
602  }
603  }
604 
605  free (
606  symbol_buffer); // NOLINT(cppcoreguidelines-owning-memory,cppcoreguidelines-no-malloc)
607 
608 #elif defined(__GNUC__) || defined(__clang__)
609  // POSIX: backtrace () for the addresses, dladdr () for the exported
610  // symbol names, the module and the address inside it for the rest.
611  // No process is started and no debug information is read: a frame
612  // costs microseconds, and the module offsets are symbolized offline
613  // from the module's debug file.
614  // NOTE: backtrace () is a GNU extension, not in POSIX.1-2024, but
615  // widely available.
616 
617 #if __has_include(<execinfo.h>)
618  void *addresses[kMaxStackFrames];
619 
620  int frame_count = backtrace (addresses, kMaxStackFrames);
621 
622  if (frame_count <= skip_frames)
623  return " Stack trace empty (insufficient frames)\n";
624 
625  int actual_frames = std::min (frame_count - skip_frames, max_frames);
626 
627  for (int i = 0; i < actual_frames; ++i)
628  result += Detail::format_frame (i, addresses[i + skip_frames]);
629 
630 #else
631  // No backtrace support - return basic info
632  result = " Stack trace unavailable (<execinfo.h> not available)\n";
633  result += lumex::core::string::utility::stringify (
634  " Current function: ", LUMEX_FUNCTION_NAME, "\n");
635 #endif
636 
637 #else
638  // Unknown platform
639  result = " Stack trace not supported on this platform\n";
640  result += lumex::core::string::utility::stringify (
641  " Current function: ", LUMEX_FUNCTION_NAME, "\n");
642 #endif
643  }
644  catch (std::exception const &exc)
645  {
646  return lumex::core::string::utility::stringify (
647  " Stack trace unavailable (exception: ", exc.what (), ")\n");
648  }
649  catch (...)
650  {
651  return " Stack trace unavailable (unknown exception)\n";
652  }
653 
654  return result.empty () ? " Stack trace empty\n" : result;
655 }
