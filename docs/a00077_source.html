209template <typename CharT, typename Traits>
210void
211print_serial_ports_info (std::basic_ostream<CharT, Traits> &os,
212 std::vector<serial_port_info_t> const &infos)
213{
214 std::vector<serial_port_info_t const *> by_free;
215 std::vector<serial_port_info_t const *> by_busy;
216 std::vector<serial_port_info_t const *> by_available;
217
218 for (std::vector<serial_port_info_t>::const_iterator it = infos.begin ();
219 it != infos.end (); ++it)
220 {
221 if (it->state == serial_port_state::free)
222 by_free.push_back (&(*it));
223 else if (it->state == serial_port_state::busy)
224 by_busy.push_back (&(*it));
225 else
226 by_available.push_back (&(*it));
227 }
228
229 os << "\n=== Ports: available (device connected, port free) ===\n";
230 for (std::size_t i = 0; i < by_available.size (); ++i)
231 {
232 serial_port_info_t const &info = *by_available[i];
233 os << " " << info.path.c_str ();
234 if (!info.friendly_name.empty ())
235 os << " [" << info.friendly_name.c_str () << "]";
236 if (!info.vid.empty () || !info.pid.empty ())
237 os << " VID:" << info.vid.c_str () << " PID:" << info.pid.c_str ();
238 if (!info.manufacturer.empty ())
239 os << " Mfg:" << info.manufacturer.c_str ();
240 if (!info.product.empty ())
241 os << " Product:" << info.product.c_str ();
242 if (!info.serial_number.empty ())
243 os << " SN:" << info.serial_number.c_str ();
244 if (!info.driver.empty ())
245 os << " Driver:" << info.driver.c_str ();
246 if (!info.bus_info.empty ())
247 os << " Bus:" << info.bus_info.c_str ();
248 if (info.vid.empty () && info.manufacturer.empty ()
249 && !info.device_details.empty ())
250 os << " " << info.device_details.c_str ();
251 if (!info.system_error.empty ())
252 os << " Error:" << info.system_error.c_str ();
253 os << "\n";
254 }
255
256 os << "\n=== Ports: busy (used by another process) ===\n";
257 for (std::size_t i = 0; i < by_busy.size (); ++i)
258 {
259 serial_port_info_t const &info = *by_busy[i];
260 os << " " << info.path.c_str ();
261 if (!info.friendly_name.empty ())
262 os << " [" << info.friendly_name.c_str () << "]";
263 if (!info.holder_process_info.empty ())
264 os << " | " << info.holder_process_info.c_str ();
265 os << "\n";
266 }
267
268 os << "\n=== Ports: free (no physical connection) ===\n";
269 for (std::size_t i = 0; i < by_free.size (); ++i)
270 {
271 serial_port_info_t const &info = *by_free[i];
272 os << " " << info.path.c_str ();
273 os << "\n";
274 }
275
276 os << "(total: " << infos.size () << ")\n" << std::flush;
277}
