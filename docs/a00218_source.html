647 LUMEX_CONSTEXPR_FUNCTION void
648 swap (Expected &other)
649#if __cplusplus >= 201703L
650 LUMEX_NOEXCEPT_IF (std::is_nothrow_move_constructible<ErrorType>::value
651 &&std::is_nothrow_swappable<ErrorType>::value)
652#else
653 LUMEX_NOEXCEPT_IF (std::is_nothrow_move_constructible<ErrorType>::value)
654#endif
655 {
656 if (this == &other)
657 return;
658
659 if (m_has_value && other.m_has_value)
660 {
661 // Both hold void success; nothing to do
662 }
663 else if (!m_has_value && !other.m_has_value)
664 { // Both hold an error
665 std::swap (m_storage.m_error, other.m_storage.m_error);
666 }
667 else
668 { // One holds void success, the other an error. Move is required.
669 if (m_has_value)
670 { // this holds void success, other holds an error
671 new (std::addressof (m_storage.m_error))
672 ErrorType (std::move (other.m_storage.m_error));
673 other.m_storage.m_error.~ErrorType ();
674 // other now holds void success
675 }
676 else
677 { // this holds an error, other holds void success
678 new (std::addressof (other.m_storage.m_error))
679 ErrorType (std::move (m_storage.m_error));
680 m_storage.m_error.~ErrorType ();
681 // this now holds void success
682 }
683 std::swap (m_has_value, other.m_has_value);
684 }
